===== FILE: README.md =====
# Community Map

Campus events on the AT Protocol. Students browse a feed of events near UCSD, filter by tag or org, and subscribe to get live updates. Hosts post events with their Bluesky account, so the data lives in their own PDS and not in our database.

**Status:** working prototype. It has 98 backend tests, 81 frontend tests and 16 Playwright user-story tests, and CI runs all of them.

## Run it

```bash
python3 -m venv .venv && .venv/bin/pip install -e "./backend[dev]"
cd backend
CM_DB_PATH=data/demo.db ../.venv/bin/python -m app.seed      # 12 sample UCSD events
CM_DB_PATH=data/demo.db CM_DEV_MODE=true ../.venv/bin/uvicorn app.main:app --reload
# open http://localhost:8000   (API docs: /docs)
```

`CM_DEV_MODE=true` turns on **Demo mode**: you can post events without a Bluesky account. Set `CM_JETSTREAM_ENABLED=true` to index real records from the network.

Tests: `cd backend && ../.venv/bin/pytest` · `cd frontend && npm ci && npm test`

User-story tests (Playwright; starts its own seeded server on :8100):
```bash
cd e2e && npm ci && npx playwright install chromium && npx playwright test
```

| User story | Spec |
|---|---|
| Student discovers and filters events (desktop + mobile) | `e2e/tests/discover.spec.js` |
| Last-minute Larry: moves location, attendees notified live | `e2e/tests/last-minute-larry.spec.js` |
| Host Henry: Bluesky hashtag post syncs to the event | `e2e/tests/host-henry.spec.js` |
| Events expire gracefully | `e2e/tests/expiry.spec.js` |
| Poor WiFi: cached feed while offline | `e2e/tests/offline.spec.js` |

## What it does

| Requirement (from the spec) | Where |
|---|---|
| Curated feed: list view first, filters by tag, group and text, "near me" | `frontend/src/views/listView.js`, `filterBar.js`, `GET /api/events` |
| Map of events/POIs (secondary tab, loaded on demand) | `frontend/src/views/mapView.js` (Leaflet + OSM; tile URL is configurable) |
| Hosts post events with a time window and location | `frontend/src/atproto.js` writes `app.communitymap.event` to the user's PDS |
| Events expire gracefully | Badges show "Ending soon", subscribers get an "ended" notice, then `backend/app/expiry.py` removes the event |
| Subscribe to an event and get notified when it changes | `backend/app/notify/`, SSE stream, plus catch-up on reconnect |
| Listeners: sync event updates from Bluesky posts (Host Henry) | An `app.communitymap.listener` record. A post with `#hashtag` becomes an announcement on the event |
| Import from other sources | iCal feeds (`backend/app/ingest/ical.py`, `POST /api/import/ical`) |
| Poor campus WiFi | A service worker caches the app shell and API responses, the app shows a stale-data banner, and responses are gzipped |
| No location stored | The user's position stays in the browser. The server never sees it |

from datetime import UTC, datetime
from typing import Annotated, Any

from fastapi import APIRouter, Depends, HTTPException, Query

from app.deps import get_store
from app.models import EventQuery
from app.store.base import EventStore
from app.views import event_view, update_view

router = APIRouter(prefix="/api")

Store = Annotated[EventStore, Depends(get_store)]


@router.get("/events")
def list_events(
    store: Store,
    tag: Annotated[list[str], Query(max_length=32)] = [],  # noqa: B006 (FastAPI copies it)
    group: Annotated[str | None, Query(max_length=100)] = None,
    q: Annotated[str | None, Query(max_length=200)] = None,
    before: datetime | None = None,
    after: datetime | None = None,
    limit: Annotated[int, Query(ge=1, le=100)] = 20,
    cursor: Annotated[str | None, Query(max_length=1000)] = None,
) -> dict[str, Any]:
    now = datetime.now(UTC)
    query = EventQuery(
        tags=tag,
        group=group or None,
        text=q or None,
        starts_before=before,
        ends_after=after,
        limit=limit,
        cursor=cursor or None,
    )
    try:
        page = store.query_events(query, now)
    except ValueError as e:
        raise HTTPException(status_code=400, detail="invalid cursor") from e
    return {"events": [event_view(ev, now) for ev in page.events], "cursor": page.cursor}


@router.get("/event")
def get_event(uri: str, store: Store) -> dict[str, Any]:
    ev = store.get_event(uri)
    if ev is None:
        raise HTTPException(status_code=404, detail="event not found")
    view = event_view(ev, datetime.now(UTC))
    view["updates"] = [update_view(u) for u in store.list_updates(uri)]
    return view


@router.get("/tags")
def list_tags(store: Store) -> list[dict[str, Any]]:
    return [c.model_dump() for c in store.list_tags(datetime.now(UTC))]


@router.get("/groups")
def list_groups(store: Store) -> list[dict[str, Any]]:
    return [c.model_dump() for c in store.list_groups(datetime.now(UTC))]

"""POST /api/import/ical: import events from an iCalendar feed on demand."""

from __future__ import annotations

from collections.abc import AsyncIterator, Callable
from typing import Annotated
from urllib.parse import urlsplit

import httpx
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, field_validator

from app.deps import get_notifier, get_store
from app.ingest.ical import import_feed
from app.ingest.netguard import is_public_url
from app.notify.base import Notifier
from app.store.base import EventStore

router = APIRouter(prefix="/api")


class IcalImportRequest(BaseModel):
    url: str

    @field_validator("url")
    @classmethod
    def _http_only(cls, v: str) -> str:
        parts = urlsplit(v.strip())
        if parts.scheme not in ("http", "https") or not parts.hostname:
            raise ValueError("url must be an http(s) URL")
        return v.strip()


class IcalImportResponse(BaseModel):
    imported: int


async def get_http_client() -> AsyncIterator[httpx.AsyncClient]:
    async with httpx.AsyncClient() as client:
        yield client


def get_url_guard() -> Callable[[str], bool]:
    return is_public_url


@router.post("/import/ical")
async def import_ical(
    body: IcalImportRequest,
    url_guard: Annotated[Callable[[str], bool], Depends(get_url_guard)],
    store: Annotated[EventStore, Depends(get_store)],
    notifier: Annotated[Notifier, Depends(get_notifier)],
    client: Annotated[httpx.AsyncClient, Depends(get_http_client)],
) -> IcalImportResponse:
    if not url_guard(body.url):
        raise HTTPException(422, "url must point to a public host")
    try:
        n = await import_feed(body.url, store, notifier, client)
    except httpx.HTTPError as e:
        raise HTTPException(502, f"could not fetch feed: {e}") from e
    except ValueError as e:
        raise HTTPException(502, "feed is not a valid iCalendar file") from e
    return IcalImportResponse(imported=n)

import asyncio
from collections.abc import AsyncIterator, Awaitable, Callable
from typing import Annotated, Any

from fastapi import APIRouter, Depends, HTTPException, Query, Request, Response
from fastapi.responses import StreamingResponse
from pydantic import BaseModel, ConfigDict, Field
from pydantic.alias_generators import to_camel

from app.deps import get_notifier, get_store
from app.models import Notification
from app.notify.base import Notifier
from app.notify.service import SubscriptionNotifier
from app.store.base import EventStore

router = APIRouter(prefix="/api")

SUBSCRIBER_ID_PATTERN = r"^[A-Za-z0-9_-]{8,64}$"
PING_INTERVAL_S = 15.0

Store = Annotated[EventStore, Depends(get_store)]
SubscriberIdQuery = Annotated[str, Query(alias="subscriberId", pattern=SUBSCRIBER_ID_PATTERN)]


class SubscriptionBody(BaseModel):
    model_config = ConfigDict(alias_generator=to_camel, populate_by_name=True)

    subscriber_id: str = Field(pattern=SUBSCRIBER_ID_PATTERN)
    event_uri: str = Field(min_length=1, max_length=500)


def _notification_json(n: Notification) -> dict[str, Any]:
    return n.model_dump(mode="json", by_alias=True)


@router.post("/subscriptions", status_code=204)
def subscribe(body: SubscriptionBody, store: Store) -> Response:
    if store.get_event(body.event_uri) is None:
        raise HTTPException(status_code=404, detail="event not found")
    store.subscribe(body.subscriber_id, body.event_uri)
    return Response(status_code=204)


@router.delete("/subscriptions", status_code=204)
def unsubscribe(body: SubscriptionBody, store: Store) -> Response:
    store.unsubscribe(body.subscriber_id, body.event_uri)
    return Response(status_code=204)


@router.get("/subscriptions")
def list_subscriptions(subscriber_id: SubscriberIdQuery, store: Store) -> list[str]:
    return store.subscriptions_of(subscriber_id)


@router.get("/notifications")
def list_notifications(
    subscriber_id: SubscriberIdQuery,
    store: Store,
    since: Annotated[int, Query(ge=0)] = 0,
) -> list[dict[str, Any]]:
    return [_notification_json(n) for n in store.list_notifications(subscriber_id, since)]


async def sse_messages(
    notifier: SubscriptionNotifier,
    subscriber_id: str,
    is_disconnected: Callable[[], Awaitable[bool]],
    ping_interval_s: float = PING_INTERVAL_S,
) -> AsyncIterator[str]:
    """Yield SSE frames for one client until it disconnects; always releases its queue."""
    queue = notifier.connect(subscriber_id)
    try:
        yield ": connected\n\n"
        while not await is_disconnected():
            try:
                n = await asyncio.wait_for(queue.get(), timeout=ping_interval_s)
            except TimeoutError:
                yield ": ping\n\n"
                continue
            yield f"id: {n.id}\ndata: {n.model_dump_json(by_alias=True)}\n\n"
    finally:
        notifier.disconnect(subscriber_id, queue)


@router.get("/notifications/stream")
async def stream_notifications(
    request: Request,
    subscriber_id: SubscriberIdQuery,
    notifier: Annotated[Notifier, Depends(get_notifier)],
) -> StreamingResponse:
    if not isinstance(notifier, SubscriptionNotifier):
        raise HTTPException(status_code=503, detail="live notifications unavailable")
    return StreamingResponse(
        sse_messages(notifier, subscriber_id, request.is_disconnected),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},
    )

===== FILE: backend/app/expiry.py =====
"""Periodic removal of ended events, notifying subscribers first."""

from __future__ import annotations

import asyncio
import logging
from datetime import UTC, datetime, timedelta

from app.config import Settings
from app.models import EventQuery, IndexedEvent
from app.notify.base import Notifier
from app.store.base import EventStore

log = logging.getLogger(__name__)

_EPOCH = datetime(1970, 1, 1, tzinfo=UTC)


def _expired_events(store: EventStore, cutoff: datetime) -> list[IndexedEvent]:
    """All events with endsAt < cutoff, found via the Protocol's query_events."""
    out: list[IndexedEvent] = []
    q = EventQuery(starts_before=cutoff, ends_after=_EPOCH, limit=100)
    while True:
        page = store.query_events(q, cutoff)
        out += [ev for ev in page.events if _utc(ev.record.ends_at) < cutoff]
        if page.cursor is None:
            return out
        q = q.model_copy(update={"cursor": page.cursor})


def _utc(dt: datetime) -> datetime:
    return dt.replace(tzinfo=UTC) if dt.tzinfo is None else dt


async def sweep_once(
    store: EventStore, notifier: Notifier, now: datetime, grace_s: int
) -> list[str]:
    cutoff = _utc(now) - timedelta(seconds=grace_s)
    for ev in _expired_events(store, cutoff):
        await notifier.publish(ev.uri, "expired", f"{ev.record.name} has ended")
    removed = store.delete_expired(cutoff)
    if removed:
        log.info("expired %d event(s)", len(removed))
    return removed

    while True:
        try:
            await sweep_once(store, notifier, datetime.now(UTC), settings.expiry_grace_s)
        except Exception:
            log.exception("expiry sweep failed")
        await asyncio.sleep(settings.expiry_interval_s)

from __future__ import annotations

import logging

from collections.abc import Callable

def did_document_url(did: str, plc_url: str = "https://plc.directory") -> str | None:
    if did.startswith("did:plc:"):
        return f"{plc_url}/{did}"
    if did.startswith("did:web:"):
        host, *path = did.removeprefix("did:web:").split(":")
        host = unquote(host)
        if not host:
            return None
        if path:
            return f"https://{host}/{'/'.join(path)}/did.json"
        return f"https://{host}/.well-known/did.json"
    return None


def handle_from_doc(doc: Any) -> str | None:
    aka = doc.get("alsoKnownAs") if isinstance(doc, dict) else None
    for entry in aka if isinstance(aka, list) else []:
        if isinstance(entry, str) and entry.startswith("at://") and len(entry) > 5:
            return entry.removeprefix("at://")
    return None


class HandleResolver:
    """Unverified handle lookup with an in-memory LRU + TTL cache. Never raises."""

    def __init__(
        self,
        client: httpx.AsyncClient | None = None,
        ttl_s: float = 3600,
        max_size: int = 10_000,
        plc_url: str = "https://plc.directory",
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        self.client = client or httpx.AsyncClient(timeout=3.0)
        self.ttl_s, self.max_size, self.plc_url, self.clock = ttl_s, max_size, plc_url, clock
        self._cache: OrderedDict[str, tuple[float, str | None]] = OrderedDict()

    async def resolve(self, did: str) -> str | None:
        hit = self._cache.get(did)
        if hit is not None and hit[0] > self.clock():
            self._cache.move_to_end(did)
            return hit[1]
        handle = await self._fetch(did)
        self._remember(did, handle)
        return handle

    async def _fetch(self, did: str) -> str | None:
        url = did_document_url(did, self.plc_url)
        if url is None:
            return None
        try:
            resp = await self.client.get(url, timeout=3.0)
            resp.raise_for_status()
            return handle_from_doc(resp.json())
        except (httpx.HTTPError, ValueError) as e:
            log.info("handle resolution failed for %s: %r", did, e)
            return None

    def _remember(self, did: str, handle: str | None) -> None:
        self._cache[did] = (self.clock() + self.ttl_s, handle)
        self._cache.move_to_end(did)
        while len(self._cache) > self.max_size:
            self._cache.popitem(last=False)

    async def aclose(self) -> None:
        await self.client.aclose()

def _sha1(s: str) -> str:
    return hashlib.sha1(s.encode()).hexdigest()


def event_uri(feed_url: str, uid: str) -> str:
    return f"ical://{_sha1(feed_url)[:12]}/{_sha1(uid)[:16]}"


def _to_utc(value: date | datetime) -> datetime:
    if isinstance(value, datetime):
        return value.replace(tzinfo=UTC) if value.tzinfo is None else value.astimezone(UTC)
    return datetime(value.year, value.month, value.day, tzinfo=UTC)  # all-day


def _window(ve: Component) -> tuple[datetime, datetime]:
    raw_start = ve["DTSTART"].dt
    start = _to_utc(raw_start)
    if "DTEND" in ve:
        end = _to_utc(ve["DTEND"].dt)
    elif "DURATION" in ve:
        end = start + ve["DURATION"].dt
    elif isinstance(raw_start, datetime):
        end = start + DEFAULT_DURATION
    else:
        end = start + timedelta(days=1)  # all-day event without DTEND
    return start, end


def _text(ve: Component, key: str, limit: int) -> str | None:
    value = ve.get(key)
    if value is None:
        return None
    return str(value).strip()[:limit] or None


def _location(ve: Component) -> Location:
    name = _text(ve, "LOCATION", 200) or NO_LOCATION
    geo = ve.get("GEO")
    if geo is None:
        return Location(name=name)
    return Location(name=name, lat=str(geo.latitude), lng=str(geo.longitude))


def _record(ve: Component, now: datetime) -> EventRecord:
    start, end = _window(ve)
    created = ve.get("CREATED")
    return EventRecord(
        name=_text(ve, "SUMMARY", 200) or "",
        description=_text(ve, "DESCRIPTION", 3000) or "",
        starts_at=start,
        ends_at=end,
        location=_location(ve),
        tags=[str(c) for c in ve.categories][:10],
        source_url=_text(ve, "URL", 2000),
        created_at=_to_utc(created.dt) if created is not None else now,
    )


def _event(ve: Component, feed_url: str, author: str, now: datetime) -> IndexedEvent | None:
    uid = _text(ve, "UID", 1000)
    if uid is None:
        raise ValueError("VEVENT without UID")
    record = _record(ve, now)
    if record.ends_at < now:
        return None
    return IndexedEvent(
        uri=event_uri(feed_url, uid),
        author_did=author,
        source="ical",
        record=record,
        indexed_at=now,
        updated_at=now,
    )


def parse_ics(text: str, feed_url: str, now: datetime) -> list[IndexedEvent]:
    """Parse a feed into IndexedEvents. Raises ValueError if text is not a calendar at all."""
    cal = Calendar.from_ical(text)
    author = "ical:" + (urlsplit(feed_url).hostname or "unknown")
    out: list[IndexedEvent] = []
    for ve in cal.walk("VEVENT"):
        try:
            ev = _event(ve, feed_url, author, now)
        except (ValueError, KeyError, TypeError, AttributeError, ValidationError) as e:
            log.info("skipping malformed VEVENT in %s: %s", feed_url, e)
            continue
        if ev is not None:
            out.append(ev)
    return out


async def import_feed(
    url: str, store: EventStore, notifier: Notifier, client: httpx.AsyncClient
) -> int:
    """Fetch and index one feed; returns the number of (current) events imported.

    Raises httpx.HTTPError on fetch failure, ValueError if the body is not iCalendar.
    """
    resp = await client.get(url, timeout=FETCH_TIMEOUT_S, follow_redirects=True)
    resp.raise_for_status()
    events = parse_ics(resp.text, url, utcnow())
    for ev in events:
        existing = store.get_event(ev.uri)
        if existing is not None:  # feeds rarely carry CREATED; keep ours stable
            record = ev.record.model_copy(update={"created_at": existing.record.created_at})
            ev = ev.model_copy(update={"record": record})
        await index_event(store, notifier, ev, existing)
    return len(events)

            for url in urls:

            await http.aclose()

"""Jetstream (JSON firehose) client: feeds commit messages to a CommitProcessor.

    return None if stored is None else max(stored - REWIND_US, 0)


class Backoff:
    """Exponential backoff: 1s, 2s, 4s ... capped at 60s."""

    def __init__(self, initial: float = 1.0, cap: float = 60.0) -> None:
        self.initial, self.cap = initial, cap
        self.current = initial

    def next(self) -> float:
        delay = self.current
        self.current = min(self.current * 2, self.cap)
        return delay

    def reset(self) -> None:
        self.current = self.initial


class CursorSaver:
    """Tracks the latest time_us and persists it every `every` messages or `interval_s`."""

    def __init__(
        self,
        store: EventStore,
        name: str,
        every: int = 100,
        interval_s: float = 5.0,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        self.store, self.name = store, name
        self.every, self.interval_s, self.clock = every, interval_s, clock
        self.latest: int | None = None
        self._pending = 0
        self._last_save = clock()

    def seen(self, time_us: Any) -> None:
        if not isinstance(time_us, int):
            return
        self.latest = max(time_us, self.latest or 0)
        self._pending += 1
        if self._pending >= self.every or self.clock() - self._last_save >= self.interval_s:
            self.flush()

    def flush(self) -> None:
        if self.latest is not None and self._pending:
            self.store.set_cursor(self.name, self.latest)
        self._pending = 0
        self._last_save = self.clock()


class JetstreamClient:
    """Note: installs itself as processor.on_listeners_changed to restart the posts stream."""

    def __init__(
        self,
        url: str,
        processor: CommitProcessor,
        store: EventStore,
        cursor_name: str = "jetstream",
        connect: Connect = websockets.connect,
        sleep: Sleep = asyncio.sleep,
    ) -> None:
        self.url = url
        self.processor = processor
        self.store = store
        self.cursor_name = cursor_name
        self.connect = connect
        self.sleep = sleep
        self._dids_changed = asyncio.Event()
        processor.on_listeners_changed = self.listeners_changed

    def build_url(self, collections: list[str], dids: list[str], cursor: int | None) -> str:
        params: list[tuple[str, str]] = [("wantedCollections", c) for c in collections]
        params += [("wantedDids", d) for d in dids]
        if cursor is not None:
            params.append(("cursor", str(cursor)))
        sep = "&" if "?" in self.url else "?"
        return f"{self.url}{sep}{urlencode(params)}" if params else self.url

    def listeners_changed(self, dids: list[str]) -> None:
        self._dids_changed.set()

    async def run_forever(self) -> None:
        await asyncio.gather(
            self.run_stream(self.cursor_name, [EVENT_NSID, LISTENER_NSID], []),
            self._run_posts(),
        )

    async def run_stream(self, name: str, collections: list[str], dids: list[str]) -> None:
        """Consume one subscription forever, reconnecting with backoff."""
        backoff = Backoff()
        while True:
            try:
                if await self.consume_once(name, collections, dids):
                    backoff.reset()
            except asyncio.CancelledError:
                raise
            except Exception as e:  # network errors, bad handshakes, ...
                log.warning("jetstream %s disconnected: %r", name, e)
            await self.sleep(backoff.next())

    async def consume_once(self, name: str, collections: list[str], dids: list[str]) -> int:
        """One connection: resume from stored cursor, process until closed. Returns msg count."""
        saver = CursorSaver(self.store, name)
        url = self.build_url(collections, dids, resume_cursor(self.store.get_cursor(name)))
        count = 0
        try:
            async with self.connect(url) as ws:
                log.info("jetstream %s connected", name)
                async for raw in ws:
                    count += 1
                    await self._process(raw, saver)
        finally:
            saver.flush()
        return count

    async def _process(self, raw: str | bytes, saver: CursorSaver) -> None:
        msg = parse_message(raw)
        if msg is None:
            return
        try:
            await self.processor.handle(msg)
        except Exception:
            log.exception("failed to process jetstream message")
        saver.seen(msg.get("time_us"))

    async def _run_posts(self) -> None:
        name = f"{self.cursor_name}:posts"
        while True:
            self._dids_changed.clear()
            dids = self.store.listener_dids()[:MAX_WANTED_DIDS]
            task = None
            if dids:
                task = asyncio.create_task(self.run_stream(name, [POST_NSID], dids))
            try:
                await self._dids_changed.wait()
            finally:
                if task is not None:
                    task.cancel()
                    with contextlib.suppress(asyncio.CancelledError):
                        await task

"""SSRF guard: only allow fetching URLs whose host resolves to public IP addresses."""

from __future__ import annotations

import ipaddress
import socket
from urllib.parse import urlsplit


def is_public_url(url: str, resolve=socket.getaddrinfo) -> bool:
    parts = urlsplit(url)
    if parts.scheme not in ("http", "https") or not parts.hostname:
        return False
    try:
        infos = resolve(parts.hostname, parts.port or 443, proto=socket.IPPROTO_TCP)
    except (OSError, UnicodeError):
        return False
    addrs = {ipaddress.ip_address(info[4][0].split("%")[0]) for info in infos}
    return bool(addrs) and all(a.is_global for a in addrs)

def parse_commit(msg: dict[str, Any]) -> Commit | None:

        return None

    coll, rkey = body.get("collection"), body.get("rkey")

        return None
    record = body.get("record")
    return Commit(
        did, op, coll, rkey, body.get("cid"), record if isinstance(record, dict) else None
    )

def _fmt_time(dt: datetime) -> str:

def _fmt_list(items: list[str]) -> str:
    return ", ".join(items) or "none"


def diff_events(old: EventRecord, new: EventRecord) -> list[str]:
    """Human-readable list of what changed between two versions of an event."""
    out: list[str] = []
    if old.name != new.name:
        out.append(f"Name changed: {old.name} → {new.name}")
    if old.starts_at != new.starts_at:
        out.append(f"Time changed: {_fmt_time(old.starts_at)} → {_fmt_time(new.starts_at)}")
    elif old.ends_at != new.ends_at:
        out.append(f"End time changed: {_fmt_time(old.ends_at)} → {_fmt_time(new.ends_at)}")
    if old.location.name != new.location.name:
        out.append(f"Location changed: {old.location.name} → {new.location.name}")
    elif old.location != new.location:
        out.append("Location details updated")
    if old.description != new.description:
        out.append("Description updated")
    if old.tags != new.tags:
        out.append(f"Tags changed: {_fmt_list(old.tags)} → {_fmt_list(new.tags)}")
    if old.group != new.group:
        out.append(f"Group changed: {old.group or 'none'} → {new.group or 'none'}")
    return out


async def index_event(
    store: EventStore, notifier: Notifier, ev: IndexedEvent, existing: IndexedEvent | None
) -> UpsertResult:
    """Upsert ev (keeping existing.indexed_at) and notify subscribers of meaningful changes."""
    if existing is not None:
        ev = ev.model_copy(update={"indexed_at": existing.indexed_at})
    result = store.upsert_event(ev)
    if result == "updated" and existing is not None:
        diffs = diff_events(existing.record, ev.record)
        if diffs:
            await notifier.publish(ev.uri, "updated", "; ".join(diffs))
    return result

def extract_hashtags(post: dict[str, Any]) -> set[str]:
    """Lower-cased hashtags from a post's text and its richtext tag facets."""
    text = post.get("text")
    tags = set()
    if isinstance(text, str):
        tags = {m.rstrip(_TRAILING_PUNCT).lower() for m in _HASHTAG.findall(text)}
    for facet in post.get("facets") or []:
        for feat in facet.get("features", []) if isinstance(facet, dict) else []:
            if isinstance(feat, dict) and feat.get("$type") == TAG_FACET:
                tag = feat.get("tag")
                if isinstance(tag, str):
                    tags.add(tag.lstrip("#").lower())
    tags.discard("")
    return tags

class CommitProcessor:
    def __init__(
        self,
        store: EventStore,
        notifier: Notifier,
        handle_resolver: HandleLookup | None = None,
        on_listeners_changed: ListenersChanged | None = None,
        clock: Callable[[], datetime] = utcnow,
    ) -> None:
        self.store = store
        self.notifier = notifier
        self.handle_resolver = handle_resolver
        self.on_listeners_changed = on_listeners_changed
        self.clock = clock

    async def handle(self, msg: dict[str, Any]) -> None:
        c = parse_commit(msg)
        if c is None:
            return
        if c.collection == EVENT_NSID:
            await self._event(c)
        elif c.collection == LISTENER_NSID:
            await self._listener(c)
        elif c.collection == POST_NSID and c.operation == "create":
            await self._post(c)

    # events

    async def _event(self, c: Commit) -> None:
        if c.operation == "delete":
            await self._delete_event(c.uri)
            return
        try:
            record = EventRecord.model_validate(c.record or {})
        except ValidationError as e:
            log.info("skipping invalid event %s: %s", c.uri, e.errors()[:3])
            return
        existing = self.store.get_event(c.uri)
        now = self.clock()
        ev = IndexedEvent(
            uri=c.uri,
            cid=c.cid,
            author_did=c.did,
            author_handle=await self._handle(c.did, existing),
            source="atproto",
            record=record,
            indexed_at=now,
            updated_at=now,
        )
        await index_event(self.store, self.notifier, ev, existing)

    async def _delete_event(self, uri: str) -> None:
        existing = self.store.get_event(uri)
        if existing is None:
            return
        # Publish first: the notifier looks up subscribers of this event.
        await self.notifier.publish(uri, "deleted", f"{existing.record.name} was cancelled")
        self.store.delete_event(uri)

    async def _handle(self, did: str, existing: IndexedEvent | None) -> str | None:
        if existing is not None and existing.author_handle:
            return existing.author_handle
        if self.handle_resolver is None:
            return None
        return await self.handle_resolver.resolve(did)

    # listeners

    async def _listener(self, c: Commit) -> None:
        before = self.store.listener_dids()
        if c.operation == "delete":
            self.store.delete_listener(c.uri)
        else:
            listener = self._parse_listener(c)
            if listener is None:
                return
            self.store.upsert_listener(listener)
        after = self.store.listener_dids()
        if after != before and self.on_listeners_changed is not None:
            res = self.on_listeners_changed(after)
            if inspect.isawaitable(res):
                await res

    def _parse_listener(self, c: Commit) -> Listener | None:
        try:
            record = ListenerRecord.model_validate(c.record or {})
        except ValidationError as e:
            log.info("skipping invalid listener %s: %s", c.uri, e.errors()[:3])
            return None
        if not record.subject.startswith(f"at://{c.did}/{EVENT_NSID}/"):
            log.info("rejecting listener %s: subject %s not owned by author", c.uri, record.subject)
            return None
        return Listener(uri=c.uri, author_did=c.did, record=record)

    # posts

    async def _post(self, c: Commit) -> None:
        listeners = self.store.listeners_for_did(c.did)
        if not listeners or c.record is None:
            return
        tags = extract_hashtags(c.record)
        subjects = {li.record.subject for li in listeners if li.record.hashtag in tags}
        text = c.record.get("text") or ""
        for subject in sorted(subjects):
            upd = EventUpdate(
                event_uri=subject, text=text, source_uri=c.uri, created_at=self.clock()
            )
            self.store.add_update(upd)
            await self.notifier.publish(subject, "announcement", text)

def create_app(settings: Settings | None = None, *, background: bool = True) -> FastAPI:
    settings = settings or get_settings()

    @asynccontextmanager
    async def lifespan(app: FastAPI) -> AsyncIterator[None]:
        store = SqliteStore(settings.db_path)
        notifier = SubscriptionNotifier(store)
        resolver = HandleResolver()
        processor = CommitProcessor(store, notifier, resolver)
        app.state.settings, app.state.store = settings, store
        app.state.notifier, app.state.processor = notifier, processor

        tasks: list[asyncio.Task] = []
        if background:
            tasks.append(asyncio.create_task(run_expiry_loop(store, notifier, settings)))
            tasks.append(asyncio.create_task(run_ical_loop(store, notifier, settings)))
            if settings.jetstream_enabled:
                client = JetstreamClient(settings.jetstream_url, processor, store)
                tasks.append(asyncio.create_task(client.run_forever()))
        try:
            yield
        finally:
            for t in tasks:
                t.cancel()
            for t in tasks:
                with contextlib.suppress(asyncio.CancelledError):
                    await t
            await resolver.aclose()
            store.close()

    app = FastAPI(title="Community Map AppView", version="0.1.0", lifespan=lifespan)
    app.add_middleware(GZipMiddleware, minimum_size=1024)
    app.add_middleware(
        CORSMiddleware,
        allow_origins=[o.strip() for o in settings.cors_origins.split(",") if o.strip()],
        allow_methods=["*"],
        allow_headers=["*"],
    )
    for module in (meta, events, subscriptions, imports, dev):
        app.include_router(module.router)

    frontend = Path(settings.frontend_dir)
    if frontend.is_dir():
        app.mount("/", StaticFiles(directory=frontend, html=True), name="frontend")
    else:
        log.warning("frontend dir %s not found; serving API only", frontend)
    return app

"""Notifier that persists one Notification per subscriber and pushes to live SSE queues."""

from __future__ import annotations

import asyncio
import logging
from datetime import UTC, datetime

from app.models import Notification, NotificationKind
from app.store.base import EventStore

log = logging.getLogger(__name__)

QUEUE_MAXSIZE = 100


class SubscriptionNotifier:
    def __init__(self, store: EventStore) -> None:
        self._store = store
        self._queues: dict[str, set[asyncio.Queue[Notification]]] = {}

    async def publish(self, event_uri: str, kind: NotificationKind, message: str) -> int:
        now = datetime.now(UTC)
        subscribers = self._store.subscribers_of(event_uri)
        for sub in subscribers:
            n = self._store.add_notification(
                Notification(
                    subscriber_id=sub,
                    event_uri=event_uri,
                    kind=kind,
                    message=message,
                    created_at=now,
                )
            )
            self._push(sub, n)
        return len(subscribers)

    def _push(self, subscriber_id: str, n: Notification) -> None:
        for queue in self._queues.get(subscriber_id, ()):
            try:
                queue.put_nowait(n)
            except asyncio.QueueFull:
                log.warning("dropping live notification for slow client %s", subscriber_id)

    def connect(self, subscriber_id: str) -> asyncio.Queue[Notification]:
        queue: asyncio.Queue[Notification] = asyncio.Queue(maxsize=QUEUE_MAXSIZE)
        self._queues.setdefault(subscriber_id, set()).add(queue)
        return queue

    def disconnect(self, subscriber_id: str, queue: asyncio.Queue[Notification]) -> None:
        queues = self._queues.get(subscriber_id)
        if queues is None:
            return
        queues.discard(queue)
        if not queues:
            del self._queues[subscriber_id]

def to_iso(dt: datetime) -> str:
    """Fixed-width UTC ISO string (lexicographic order == chronological). Naive = UTC."""
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=UTC)
    return dt.astimezone(UTC).isoformat(timespec="microseconds")


def from_iso(s: str) -> datetime:
    return datetime.fromisoformat(s)


def encode_cursor(starts_at: str, uri: str) -> str:
    return base64.urlsafe_b64encode(f"{starts_at}|{uri}".encode()).decode()


def decode_cursor(cursor: str) -> tuple[str, str]:
    """Raises ValueError on a malformed cursor."""
    try:
        raw = base64.urlsafe_b64decode(cursor.encode()).decode()
    except (binascii.Error, UnicodeDecodeError) as e:
        raise ValueError("invalid cursor") from e
    starts_at, sep, uri = raw.partition("|")
    if not sep or not uri:
        raise ValueError("invalid cursor")
    return to_iso(from_iso(starts_at)), uri


def _norm_tag(tag: str) -> str:
    return tag.strip().lstrip("#").lower()


def _search_text(rec: EventRecord) -> str:
    return "\n".join([rec.name, rec.description, rec.location.name]).lower()


def _record_json(rec: EventRecord) -> str:
    return rec.model_dump_json(by_alias=True)


class SqliteStore:
    def __init__(self, path: str) -> None:
        if path != ":memory:":
            os.makedirs(os.path.dirname(os.path.abspath(path)), exist_ok=True)
        self._conn = sqlite3.connect(path, check_same_thread=False)
        self._conn.row_factory = sqlite3.Row
        self._lock = threading.RLock()
        self._conn.execute("PRAGMA journal_mode=WAL")
        self._conn.executescript(SCHEMA)

    def close(self) -> None:
        with self._lock:
            self._conn.close()

    @contextmanager
    def _tx(self) -> Iterator[sqlite3.Connection]:
        with self._lock, self._conn:
            yield self._conn

    def _all(self, sql: str, params: tuple[Any, ...] | list[Any] = ()) -> list[sqlite3.Row]:
        with self._lock:
            return self._conn.execute(sql, params).fetchall()

    def _one(self, sql: str, params: tuple[Any, ...] = ()) -> sqlite3.Row | None:
        with self._lock:
            return self._conn.execute(sql, params).fetchone()

    # events

    def upsert_event(self, ev: IndexedEvent) -> UpsertResult:
        rec = ev.record
        rec_json = _record_json(rec)
        with self._tx() as c:
            row = c.execute(
                "SELECT cid, record, indexed_at FROM events WHERE uri = ?", (ev.uri,)
            ).fetchone()
            if row is not None and row["cid"] == ev.cid and row["record"] == rec_json:
                return "unchanged"
            indexed_at = row["indexed_at"] if row is not None else to_iso(ev.indexed_at)
            c.execute(
                "INSERT OR REPLACE INTO events (uri, cid, author_did, author_handle, source,"
                " record, search, starts_at, ends_at, grp, indexed_at, updated_at)"
                " VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
                (
                    ev.uri,
                    ev.cid,
                    ev.author_did,
                    ev.author_handle,
                    ev.source,
                    rec_json,
                    _search_text(rec),
                    to_iso(rec.starts_at),
                    to_iso(rec.ends_at),
                    rec.group,
                    indexed_at,
                    to_iso(ev.updated_at),
                ),
            )
            c.execute("DELETE FROM event_tags WHERE uri = ?", (ev.uri,))
            c.executemany(
                "INSERT OR IGNORE INTO event_tags (uri, tag) VALUES (?, ?)",
                [(ev.uri, t) for t in rec.tags],
            )
        return "created" if row is None else "updated"

    def get_event(self, uri: str) -> IndexedEvent | None:
        row = self._one("SELECT * FROM events WHERE uri = ?", (uri,))
        return _row_to_event(row) if row is not None else None

    def delete_event(self, uri: str) -> bool:
        """Also drops tags, updates and subscriptions: publish "deleted" before calling."""
        with self._tx() as c:
            existed = c.execute("SELECT 1 FROM events WHERE uri = ?", (uri,)).fetchone()
            _purge(c, [uri])
        return existed is not None

    def query_events(self, q: EventQuery, now: datetime) -> Page:
        where = ["ends_at >= ?"]
        params: list[Any] = [to_iso(q.ends_after or now)]
        tags = [t for t in (_norm_tag(t) for t in q.tags) if t]
        if tags:
            marks = ",".join("?" * len(tags))
            where.append(
                f"EXISTS (SELECT 1 FROM event_tags t WHERE t.uri = e.uri AND t.tag IN ({marks}))"
            )
            params += tags
        if q.group:
            where.append("grp = ?")
            params.append(q.group)
        if q.text:
            where.append("instr(search, ?) > 0")
            params.append(q.text.lower())
        if q.starts_before:
            where.append("starts_at < ?")
            params.append(to_iso(q.starts_before))
        if q.cursor:
            after_start, after_uri = decode_cursor(q.cursor)
            where.append("(starts_at > ? OR (starts_at = ? AND uri > ?))")
            params += [after_start, after_start, after_uri]
        sql = f"SELECT * FROM events e WHERE {' AND '.join(where)} ORDER BY starts_at, uri LIMIT ?"
        rows = self._all(sql, [*params, q.limit + 1])
        more = len(rows) > q.limit
        rows = rows[: q.limit]
        cursor = encode_cursor(rows[-1]["starts_at"], rows[-1]["uri"]) if more else None
        return Page(events=[_row_to_event(r) for r in rows], cursor=cursor)

    def delete_expired(self, cutoff: datetime) -> list[str]:
        with self._tx() as c:
            uris = [
                r["uri"]
                for r in c.execute("SELECT uri FROM events WHERE ends_at < ?", (to_iso(cutoff),))
            ]
            _purge(c, uris)
        return uris

    def list_tags(self, now: datetime) -> list[Count]:
        return self._counts(
            "SELECT t.tag AS value, COUNT(*) AS n FROM event_tags t"
            " JOIN events e ON e.uri = t.uri WHERE e.ends_at >= ?"
            " GROUP BY t.tag ORDER BY n DESC, value",
            now,
        )

    def list_groups(self, now: datetime) -> list[Count]:
        return self._counts(
            "SELECT grp AS value, COUNT(*) AS n FROM events"
            " WHERE ends_at >= ? AND grp IS NOT NULL AND grp != ''"
            " GROUP BY grp ORDER BY n DESC, value",
            now,
        )

    def _counts(self, sql: str, now: datetime) -> list[Count]:
        return [Count(value=r["value"], count=r["n"]) for r in self._all(sql, (to_iso(now),))]

    # updates

    def add_update(self, upd: EventUpdate) -> None:
        with self._tx() as c:
            c.execute(
                "INSERT INTO updates (event_uri, text, source_uri, created_at) VALUES (?, ?, ?, ?)",
                (upd.event_uri, upd.text, upd.source_uri, to_iso(upd.created_at)),
            )

    def list_updates(self, event_uri: str) -> list[EventUpdate]:
        rows = self._all(
            "SELECT * FROM updates WHERE event_uri = ? ORDER BY created_at, id", (event_uri,)
        )
        return [
            EventUpdate(
                event_uri=r["event_uri"],
                text=r["text"],
                source_uri=r["source_uri"],
                created_at=from_iso(r["created_at"]),
            )
            for r in rows
        ]

    # listeners

    def upsert_listener(self, listener: Listener) -> None:
        with self._tx() as c:
            c.execute(
                "INSERT OR REPLACE INTO listeners (uri, author_did, record) VALUES (?, ?, ?)",
                (listener.uri, listener.author_did, listener.record.model_dump_json(by_alias=True)),
            )

    def delete_listener(self, uri: str) -> bool:
        with self._tx() as c:
            return c.execute("DELETE FROM listeners WHERE uri = ?", (uri,)).rowcount > 0

    def listeners_for_did(self, did: str) -> list[Listener]:
        rows = self._all("SELECT * FROM listeners WHERE author_did = ? ORDER BY uri", (did,))
        return [
            Listener(
                uri=r["uri"],
                author_did=r["author_did"],
                record=ListenerRecord.model_validate_json(r["record"]),
            )
            for r in rows
        ]

    def listener_dids(self) -> list[str]:
        rows = self._all("SELECT DISTINCT author_did FROM listeners ORDER BY author_did")
        return [r["author_did"] for r in rows]

    # subscriptions

    def subscribe(self, subscriber_id: str, event_uri: str) -> None:
        with self._tx() as c:
            c.execute(
                "INSERT OR IGNORE INTO subscriptions (subscriber_id, event_uri) VALUES (?, ?)",
                (subscriber_id, event_uri),
            )

    def unsubscribe(self, subscriber_id: str, event_uri: str) -> None:
        with self._tx() as c:
            c.execute(
                "DELETE FROM subscriptions WHERE subscriber_id = ? AND event_uri = ?",
                (subscriber_id, event_uri),
            )

    def subscribers_of(self, event_uri: str) -> list[str]:
        rows = self._all(
            "SELECT subscriber_id FROM subscriptions WHERE event_uri = ? ORDER BY subscriber_id",
            (event_uri,),
        )
        return [r["subscriber_id"] for r in rows]

    def subscriptions_of(self, subscriber_id: str) -> list[str]:
        rows = self._all(
            "SELECT event_uri FROM subscriptions WHERE subscriber_id = ? ORDER BY event_uri",
            (subscriber_id,),
        )
        return [r["event_uri"] for r in rows]

    # notifications

    def add_notification(self, n: Notification) -> Notification:
        with self._tx() as c:
            cur = c.execute(
                "INSERT INTO notifications (subscriber_id, event_uri, kind, message, created_at)"
                " VALUES (?, ?, ?, ?, ?)",
                (n.subscriber_id, n.event_uri, n.kind, n.message, to_iso(n.created_at)),
            )
        return n.model_copy(update={"id": cur.lastrowid})

    def list_notifications(self, subscriber_id: str, since_id: int = 0) -> list[Notification]:
        rows = self._all(
            "SELECT * FROM notifications WHERE subscriber_id = ? AND id > ? ORDER BY id",
            (subscriber_id, since_id),
        )
        return [
            Notification(
                id=r["id"],
                subscriber_id=r["subscriber_id"],
                event_uri=r["event_uri"],
                kind=r["kind"],
                message=r["message"],
                created_at=from_iso(r["created_at"]),
            )
            for r in rows
        ]

    # ingest cursors

    def get_cursor(self, name: str) -> int | None:
        row = self._one("SELECT value FROM cursors WHERE name = ?", (name,))
        return row["value"] if row is not None else None

    def set_cursor(self, name: str, value: int) -> None:
        with self._tx() as c:
            c.execute("INSERT OR REPLACE INTO cursors (name, value) VALUES (?, ?)", (name, value))


def _purge(c: sqlite3.Connection, uris: list[str]) -> None:
    params = [(u,) for u in uris]
    for table, col in (
        ("events", "uri"),
        ("event_tags", "uri"),
        ("updates", "event_uri"),
        ("subscriptions", "event_uri"),
    ):
        c.executemany(f"DELETE FROM {table} WHERE {col} = ?", params)


def _row_to_event(row: sqlite3.Row) -> IndexedEvent:
    return IndexedEvent(
        uri=row["uri"],
        cid=row["cid"],
        author_did=row["author_did"],
        author_handle=row["author_handle"],
        source=row["source"],
        record=EventRecord.model_validate(json.loads(row["record"])),
        indexed_at=from_iso(row["indexed_at"]),
        updated_at=from_iso(row["updated_at"]),
    )

from __future__ import annotations

from datetime import UTC, datetime, timedelta
from typing import Any, Literal

from app.models import EventUpdate, IndexedEvent

ENDING_SOON = timedelta(hours=1)

Status = Literal["upcoming", "live", "ending_soon"]


def _utc(dt: datetime) -> datetime:
    return dt.replace(tzinfo=UTC) if dt.tzinfo is None else dt


def event_status(ev: IndexedEvent, now: datetime) -> Status:
    now = _utc(now)
    if now < _utc(ev.record.starts_at):
        return "upcoming"
    if _utc(ev.record.ends_at) - now <= ENDING_SOON:
        return "ending_soon"
    return "live"


def event_view(ev: IndexedEvent, now: datetime) -> dict[str, Any]:
    record = ev.record.model_dump(mode="json", by_alias=True)
    meta = ev.model_dump(mode="json", by_alias=True, exclude={"record"})
    return {**meta, **record, "status": event_status(ev, now)}


def update_view(upd: EventUpdate) -> dict[str, Any]:
    return upd.model_dump(mode="json", by_alias=True, include={"text", "source_uri", "created_at"})

"""In-memory fakes of the EventStore / Notifier Protocols used by the ingest tests."""

from __future__ import annotations

from datetime import UTC, datetime

from app.models import EventUpdate, IndexedEvent, Listener, NotificationKind
from app.store.base import UpsertResult


class FakeStore:
    def __init__(self) -> None:
        self.events: dict[str, IndexedEvent] = {}
        self.updates: list[EventUpdate] = []
        self.listeners: dict[str, Listener] = {}
        self.cursors: dict[str, int] = {}

    # events
    def upsert_event(self, ev: IndexedEvent) -> UpsertResult:
        old = self.events.get(ev.uri)
        if old is not None and _same(old, ev):
            return "unchanged"
        self.events[ev.uri] = ev
        return "created" if old is None else "updated"

    def get_event(self, uri: str) -> IndexedEvent | None:
        return self.events.get(uri)

    def delete_event(self, uri: str) -> bool:
        return self.events.pop(uri, None) is not None

    # updates
    def add_update(self, upd: EventUpdate) -> None:
        self.updates.append(upd)

    def list_updates(self, event_uri: str) -> list[EventUpdate]:
        return [u for u in self.updates if u.event_uri == event_uri]

    # listeners
    def upsert_listener(self, listener: Listener) -> None:
        self.listeners[listener.uri] = listener

    def delete_listener(self, uri: str) -> bool:
        return self.listeners.pop(uri, None) is not None

    def listeners_for_did(self, did: str) -> list[Listener]:
        return [li for li in self.listeners.values() if li.author_did == did]

    def listener_dids(self) -> list[str]:
        return sorted({li.author_did for li in self.listeners.values()})

    # cursors
    def get_cursor(self, name: str) -> int | None:
        return self.cursors.get(name)

    def set_cursor(self, name: str, value: int) -> None:
        self.cursors[name] = value


def _same(a: IndexedEvent, b: IndexedEvent) -> bool:
    return (a.record, a.cid, a.author_handle) == (b.record, b.cid, b.author_handle)


class FakeNotifier:
    """Records publish calls; with a store, also records whether the event still existed."""

    def __init__(self, store: FakeStore | None = None) -> None:
        self.store = store
        self.published: list[tuple[str, NotificationKind, str]] = []
        self.event_present: list[bool] = []

    async def publish(self, event_uri: str, kind: NotificationKind, message: str) -> int:
        self.published.append((event_uri, kind, message))
        if self.store is not None:
            self.event_present.append(self.store.get_event(event_uri) is not None)
        return 1


class FakeResolver:
    def __init__(self, handles: dict[str, str]) -> None:
        self.handles = handles
        self.calls: list[str] = []

    async def resolve(self, did: str) -> str | None:
        self.calls.append(did)
        return self.handles.get(did)


def utc(*args: int) -> datetime:
    return datetime(*args, tzinfo=UTC)

class FakeConnect:
    """Each call pops a script: a list of raw messages, or an Exception to raise on connect."""

    def __init__(self, scripts: list[list[str] | Exception]) -> None:
        self.scripts = scripts
        self.urls: list[str] = []

    def __call__(self, url: str) -> Any:
        self.urls.append(url)
        script = self.scripts.pop(0) if self.scripts else []

        @asynccontextmanager
        async def cm() -> AsyncIterator[AsyncIterator[str]]:
            if isinstance(script, Exception):
                raise script

            async def gen() -> AsyncIterator[str]:
                for m in script:
                    yield m

            yield gen()

        return cm()


class StopAfter:
    """Fake sleep: records delays, cancels the loop after n sleeps."""

    def __init__(self, n: int) -> None:
        self.n = n
        self.delays: list[float] = []

    async def __call__(self, delay: float) -> None:
        self.delays.append(delay)
        await asyncio.sleep(0)  # yield like a real sleep
        if len(self.delays) >= self.n:
            raise asyncio.CancelledError


def make_client(
    connect: FakeConnect, sleep: StopAfter | None = None
) -> tuple[JetstreamClient, FakeStore]:
    store = FakeStore()

    return client, store

## How data moves

1. **Write.** The host's browser writes an `app.communitymap.event` record straight to their PDS. Our server never holds their credentials.
2. **Index.** The PDS passes the record to the relay and then to Jetstream. `JetstreamClient` sends it to `CommitProcessor`, which validates it against the lexicon and upserts it into SQLite. A cursor is saved so a restart picks up where it left off.
3. **Notify.** If an update changes something people can see (time, place, name), the processor builds a readable diff and sends it to everyone subscribed. They get it on the SSE stream, and it is also saved so they can catch up after going offline.
4. **Listeners (Host Henry).** A host can attach a listener record (event + hashtag) to an event. When that host posts on Bluesky with the hashtag, the post becomes an announcement on the event. We only subscribe to posts from accounts that have a listener, never the full firehose.
5. **Expire.** Each sweep tells subscribers that the event has ended, then deletes it from the index. The record stays on the user's PDS, since that data belongs to them.

# Design decisions to debate

These are the choices made in this prototype that go beyond the ADRs, with the main argument against each.

1. **The backend is a read-only AppView. Clients write to their own PDS.**
   *For:* the data really belongs to users, we hold no credentials, and the design follows AT Protocol conventions.
   *Against:* new events take Relay → Jetstream time (usually about a second) to show up, and we can't moderate them before they are published.

2. **Subscriptions are anonymous and server-side (a random `subscriberId` per browser), not records in the PDS.**
   *For:* subscriptions stay private (PDS records are public), and no login is needed to follow an event.
   *Against:* they don't follow you across devices, and they can't be moved to another AppView.

3. **SSE instead of Web Push.**
   *For:* no keys and no push service. It works on every browser, and missed notifications are caught up on reconnect.
   *Against:* nothing arrives while the tab is closed. Web Push (VAPID) is the next step for the Larry persona.

4. **SQLite in a single container.**
   *For:* free, nothing to operate, fast enough for one campus. The index can be rebuilt by replaying Jetstream.
   *Against:* no horizontal scaling. `EventStore` is a Protocol, so moving to Postgres means writing one new file.

5. **Expiry deletes the event from the index, not from the PDS.**
   *For:* we can't delete user data, and it shouldn't be ours to delete.
   *Against:* expired records stay on the network, and another AppView might still show them.

6. **Listeners match a hashtag on the host's own posts.**
   *For:* simple, and it can't be spoofed (the listener has to be owned by the same DID as the event).
   *Against:* the post text becomes an announcement but doesn't change the event's fields. Structured sync, such as "post says the room changed", would need parsing or an LLM.

7. **"Near me" is computed in the browser.**
   *For:* we never see or store anyone's location, which addresses the location risk in §5.3.
   *Against:* we can't do a server-side radius query, so the client re-sorts only the pages it has already loaded.

8. **Lat/lng are optional strings in the lexicon.**
   *For:* lexicons have no float type, and the list-first ADR means coordinates aren't required.
   *Against:* events without coordinates don't appear on the map. Changing a lexicon later is costly, so this is worth deciding now.

## Known gaps

- **Login uses Bluesky app passwords, not OAuth.** OAuth is the right long-term choice but needs a hosted client-metadata document.
- **There is no moderation or trust score yet.** We rely on Bluesky's account-level moderation.
- **The iCal guard can be bypassed by DNS rebinding.** Pin the resolved IP before production.
- **Bluesky login and the map tab aren't covered by the Playwright tests.** The tests use demo mode and simulate Bluesky posts with dev commits, so real login and the map are still unverified.

# How the work was split

The architect wrote the contracts first. Three engineers (Claude sub-agents) then built against those contracts in parallel. Each one owned a separate set of directories, so nobody blocked anyone else and there were no merge conflicts.

| Phase | Owner | Deliverable |
|---|---|---|
| 0. Contracts | Architect | `lexicons/`, `models.py`, `store/base.py`, `notify/base.py`, `docs/CONTRACTS.md`, repo + CI skeleton |
| 1a. Backend Core | Engineer A | SQLite store, API routers (events, subscriptions, SSE), notifier, expiry, seed data. 41 tests |
| 1b. Ingest | Engineer B | Jetstream client, commit processor + diffs, listeners, handle resolver, iCal import, dev route. 50 tests |
| 1c. Frontend | Engineer C | Vanilla-JS PWA: list/map/detail/compose views, PDS writer, demo writer, subscriptions, service worker. 81 tests |
| 2. Integration | Architect | `main.py`, end-to-end test, SSRF guard, safe defaults, Docker, docs, live checks (browser, SSE, real Jetstream) |

**Why it split this way.** The seams follow the data flow (network → index → API → UI). Each one is a Python `Protocol` or an HTTP contract. Every engineer tested against fakes of the other engineers' interfaces (for example `tests/ingest_fakes.py`), so all three could work at the same time.

**Found during integration and fixed.** Three issues came up:
- **EventView had no `createdAt`.** The Frontend needed it to edit events, so it was added.
- **The iCal import could fetch internal URLs (SSRF).** Ingest flagged this, and `netguard.py` now blocks private hosts.
- **`dev_mode` defaulted to on.** It is now off by default.

**How a human team could map to this split.** The spec lists eight developers. Lead A (backend) could take Core + Ingest with four devs. Lead B (frontend) could take the views + offline/PWA work with four devs. The contracts in Phase 0 are owned by the leads together.

/** Shared helpers for user-story specs. */

/** Card <article> for an event by name. */
export const card = (page, name) => page.locator('article.card', { hasText: name });

/** Wait until the feed has rendered at least one card. */
export async function openFeed(page) {
  await page.goto('/');
  await page.locator('article.card').first().waitFor();
}

/** ISO string `hours` from now. */
export const inHours = (hours) => new Date(Date.now() + hours * 3_600_000).toISOString();

/** datetime-local value (local time, minutes precision) `hours` from now. */
export function localInput(hours) {
  const d = new Date(Date.now() + hours * 3_600_000);
  const pad = (n) => String(n).padStart(2, '0');
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
}

/** Fill and submit the "New event" dialog in demo mode. */
export async function createEvent(page, { name, location, tags = '', hashtag = '' }) {
  await page.getByRole('button', { name: 'New event' }).click();
  const dialog = page.getByRole('dialog');
  await dialog.getByLabel('Event name').fill(name);
  await dialog.getByLabel('Starts').fill(localInput(1));
  await dialog.getByLabel('Ends').fill(localInput(3));
  await dialog.getByLabel('Location name').fill(location);
  if (tags) await dialog.getByLabel('Tags').fill(tags);
  if (hashtag) await dialog.getByLabel(/Bluesky hashtag/).fill(hashtag);
  await dialog.getByRole('button', { name: 'Create event' }).click();
  await card(page, name).waitFor();
}

/** Simulate a Jetstream commit (what the real firehose would deliver). */
export async function commit(request, did, collection, operation, rkey, record) {
  const res = await request.post('/api/dev/records', {
    data: { did, time_us: Date.now() * 1000, kind: 'commit',
      commit: { operation, collection, rkey, cid: `cid-${Date.now()}`, record } },
  });
  if (!res.ok()) throw new Error(`commit failed: ${res.status()}`);
}

/** Read the demo DID the browser is posting as. */
export const demoDid = (page) => page.evaluate(() => localStorage.getItem('cm:demo-did'));

/**
 * HTTP client for the Community Map AppView (docs/CONTRACTS.md).
 *
 * Every GET resolves to `{data, stale}`. On network failure, timeout or 5xx, a GET falls
 * back to the last cached response with `stale: true` (and `savedAt`). Writes never fall back.
 * @module api
 */
import { createCache } from './cache.js';

/** Error for non-2xx responses (and offline with nothing cached). */
export class ApiError extends Error {
  /**
   * @param {string} message
   * @param {{status?: number, body?: any, offline?: boolean}} [info]
   */
  constructor(message, { status = 0, body = null, offline = false } = {}) {
    super(message);
    this.name = 'ApiError';
    this.status = status;
    this.body = body;
    this.offline = offline;
  }
}

/**
 * Serialize params; arrays become repeated keys, empty values are dropped.
 * @param {Record<string, any>} [params]
 * @returns {string} query string including leading "?" (or "")
 */
export function toQueryString(params = {}) {
  const sp = new URLSearchParams();
  for (const [k, v] of Object.entries(params)) {
    for (const item of Array.isArray(v) ? v : [v]) {
      if (item != null && item !== '') sp.append(k, String(item));
    }
  }
  const s = sp.toString();
  return s ? `?${s}` : '';
}

/**
 * @typedef {{data: any, stale: boolean, savedAt?: number}} GetResult
 */

/**
 * Create the API client.
 * @param {{baseUrl?: string, fetchImpl?: typeof fetch, cache?: import('./cache.js').Cache,
 *          timeoutMs?: number}} [opts]
 */
export function createApi({ baseUrl = '', fetchImpl = (...a) => fetch(...a), cache = createCache(), timeoutMs = 8000 } = {}) {
  const root = baseUrl.replace(/\/$/, '');
  const url = (path, params) => `${root}${path}${toQueryString(params)}`;

  async function request(method, target, body) {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), timeoutMs);
    try {
      const init = { method, signal: ctrl.signal, headers: { Accept: 'application/json' } };
      if (body !== undefined) {
        init.headers['Content-Type'] = 'application/json';
        init.body = JSON.stringify(body);
      }
      return await fetchImpl(target, init);
    } finally {
      clearTimeout(timer);
    }
  }

  async function parse(res) {
    if (res.status === 204) return null;
    const text = await res.text();
    return text ? JSON.parse(text) : null;
  }

  async function errorFrom(res) {
    let body = null;
    try {
      body = await parse(res);
    } catch {
      /* non-JSON error body */
    }
    return new ApiError(`HTTP ${res.status}`, { status: res.status, body });
  }

  /** GET with cache write-through and stale fallback. */
  async function get(path, params, { useCache = true } = {}) {
    const target = url(path, params);
    let res;
    try {
      res = await request('GET', target);
    } catch (err) {
      return fallback(target, useCache, err);
    }
    if (res.status >= 500) return fallback(target, useCache, await errorFrom(res));
    if (!res.ok) throw await errorFrom(res);
    const data = await parse(res);
    // The service worker marks responses it served from its own cache.
    const swStale = res.headers?.get?.('x-cm-stale') === '1';
    if (useCache && !swStale) cache.set(target, data);
    return swStale ? { data, stale: true } : { data, stale: false };
  }

  function fallback(target, useCache, err) {
    const hit = useCache ? cache.get(target) : null;
    if (hit) return { data: hit.value, stale: true, savedAt: hit.savedAt };
    if (err instanceof ApiError) throw err;
    throw new ApiError('Network unavailable', { offline: true });
  }

  async function send(method, path, body) {
    let res;
    try {
      res = await request(method, url(path), body);
    } catch {
      throw new ApiError('Network unavailable', { offline: true });
    }
    if (!res.ok) throw await errorFrom(res);
    return parse(res);
  }

  return {
    /** @returns {Promise<GetResult>} `{ok:true}` */
    health: () => get('/api/health', undefined, { useCache: false }),
    /** @returns {Promise<GetResult>} `{devMode, jetstreamEnabled, eventNsid, listenerNsid}` */
    config: () => get('/api/config'),
    /**
     * @param {{tag?: string[], group?: string, q?: string, before?: string, after?: string,
     *          limit?: number, cursor?: string}} [params]
     * @returns {Promise<GetResult>} `{events, cursor}`
     */
    listEvents: (params) => get('/api/events', params),
    /** @param {string} uri @returns {Promise<GetResult>} EventView & {updates} */
    getEvent: (uri) => get('/api/event', { uri }),
    /** @returns {Promise<GetResult>} `[{value,count}]` */
    tags: () => get('/api/tags'),
    /** @returns {Promise<GetResult>} `[{value,count}]` */
    groups: () => get('/api/groups'),
    /** @param {string} subscriberId @param {string} eventUri */
    subscribe: (subscriberId, eventUri) => send('POST', '/api/subscriptions', { subscriberId, eventUri }),
    /** @param {string} subscriberId @param {string} eventUri */
    unsubscribe: (subscriberId, eventUri) => send('DELETE', '/api/subscriptions', { subscriberId, eventUri }),
    /** @param {string} subscriberId @returns {Promise<GetResult>} `[eventUri]` */
    listSubscriptions: (subscriberId) => get('/api/subscriptions', { subscriberId }),
    /**
     * @param {string} subscriberId @param {number|null} [since] notification id
     * @returns {Promise<GetResult>} `[Notification]` (never served from cache)
     */
    notifications: (subscriberId, since) =>
      get('/api/notifications', { subscriberId, since }, { useCache: false }),
    /** @param {string} subscriberId @returns {string} SSE URL for EventSource */
    notificationStreamUrl: (subscriberId) => url('/api/notifications/stream', { subscriberId }),
    /** @param {string} icalUrl @returns {Promise<{imported:number}>} */
    importIcal: (icalUrl) => send('POST', '/api/import/ical', { url: icalUrl }),
    /** @param {object} commit Jetstream-shaped commit @returns {Promise<{ok:true}>} */
    postDevRecord: (commit) => send('POST', '/api/dev/records', commit),
  };
}

/**
 * Start the app. API base comes from <meta name="cm-api-base"> (default: same origin).
 * @param {{baseUrl?: string}} [opts]
 */
export async function start({ baseUrl = $('meta[name="cm-api-base"]')?.content ?? '' } = {}) {
  const api = createApi({ baseUrl });
  const toast = createToaster($('#toasts'));
  const geo = createGeo();
  const atproto = createAtprotoWriter();
  let devWriter = null;
  const store = createStore({
    config: null, filters: { ...EMPTY_FILTERS }, events: [], cursor: null, loading: false,
    stale: false, savedAt: null, error: null, tags: [], groups: [], subscribed: new Set(),
    locating: false, tab: 'list',
  });

  const writer = () => (atproto.loggedIn ? atproto : devWriter);
  const canEdit = (ev) => Boolean(writer()) && ev.authorDid === writer().did && ev.source !== 'ical';

  // --- data loading ---
  let seq = 0;
  async function loadEvents({ append = false } = {}) {
    const mine = ++seq;
    const { filters, cursor, events } = store.get();
    store.set({ loading: true, error: null });
    try {
      const { data, stale, savedAt } = await api.listEvents(buildEventQuery(filters, { cursor: append ? cursor : null }));
      if (mine !== seq) return;
      const merged = append ? dedupe([...events, ...data.events]) : data.events;
      store.set({ events: merged, cursor: data.cursor ?? null, stale, savedAt: savedAt ?? null, loading: false });
    } catch (err) {
      if (mine !== seq) return;
      store.set({ loading: false, error: err.offline ? "You're offline and no saved events are available yet." : 'Could not load events. Please try again.' });
    }
  }

  async function loadFacets() {
    const [tags, groups] = await Promise.allSettled([api.tags(), api.groups()]);
    store.set({
      tags: tags.status === 'fulfilled' ? tags.value.data : [],
      groups: groups.status === 'fulfilled' ? groups.value.data : [],
    });
  }

  // --- filters ---
  async function changeFilters(patch) {
    const filters = { ...store.get().filters, ...patch };
    if (patch.nearMe && !geo.position) {
      store.set({ locating: true });
      try {
        await geo.request();
      } catch (err) {
        toast(err.message, { kind: 'error' });
        filters.nearMe = false;
      }
      store.set({ locating: false });
    }
    if (!filters.nearMe) geo.clear();
    store.set({ filters });
    const refetch = ['tags', 'group', 'q'].some((k) => k in patch);
    if (refetch) loadEvents();
  }

  const visibleEvents = () => {
    const { events, filters } = store.get();
    return filters.nearMe && geo.position ? sortByDistance(events, geo.position) : events;
  };

  // --- views ---
  const subs = createSubscriptions({
    api, toast,
    onChange: (subscribed) => store.set({ subscribed }),
    onNotification: (n) => applyNotification(n),
  });
  const detail = createDetailView({ api, canEdit, onEdit: (ev) => compose.open(ev), onDelete: (ev) => compose.remove(ev) });
  const compose = createComposeView({
    getWriter: writer, geo, toast,
    onSaved: () => { toast('Event saved. It can take a few seconds to appear.'); setTimeout(() => { loadEvents(); loadFacets(); }, 1500); },
    onDeleted: (uri) => { toast('Event deleted.'); store.set({ events: store.get().events.filter((e) => e.uri !== uri) }); },
  });
  const login = createLoginView({ login: (id, pw) => atproto.login(id, pw), onLoggedIn: renderAccount });
  const filterBar = createFilterBar($('#filters'), { onChange: changeFilters, geoSupported: geo.supported });
  const list = createListView($('#list'), { onLoadMore: () => loadEvents({ append: true }) });
  const map = createMapView($('#map'), { onOpen: (uri) => detail.open(uri) });

  async function toggleSubscribe(uri) {
    try {
      await subs.toggle(uri);
    } catch {
      toast('Could not update subscription — check your connection.', { kind: 'error' });
    }
  }

  async function applyNotification(n) {
    const { events } = store.get();
    if (n.kind === 'deleted' || n.kind === 'expired') {
      store.set({ events: events.filter((e) => e.uri !== n.eventUri) });
    } else if (n.kind === 'updated' && events.some((e) => e.uri === n.eventUri)) {
      try {
        const { data } = await api.getEvent(n.eventUri);
        store.set({ events: store.get().events.map((e) => (e.uri === data.uri ? data : e)) });
      } catch { /* keep old copy */ }
    }
  }

  function render() {
    const s = store.get();
    filterBar.render(s.filters, { tags: s.tags, groups: s.groups, nearMeBusy: s.locating });
    const shown = visibleEvents();
    list.render({
      events: shown, hasMore: Boolean(s.cursor), loading: s.loading, stale: s.stale, savedAt: s.savedAt,
      error: s.error, activeTags: s.filters.tags, subscribed: s.subscribed,
      onOpen: (uri) => detail.open(uri),
      onTagClick: (tag) => changeFilters({ tags: toggleTag(store.get().filters.tags, tag) }),
      onToggleSubscribe: toggleSubscribe,
    });
    if (s.tab === 'map') map.show(shown);
  }

  function renderAccount() {
    const root = $('#account');
    const canPost = Boolean(writer());
    const items = [];
    if (atproto.loggedIn) {
      items.push(h('span', { class: 'muted' }, `@${atproto.handle}`),
        h('button', { type: 'button', class: 'btn', onclick: async () => { await atproto.logout(); renderAccount(); } }, 'Log out'));
    } else {
      if (devWriter) items.push(h('span', { class: 'badge', title: 'Posts go to the local dev index only' }, 'Demo mode'));
      items.push(h('button', { type: 'button', class: 'btn', onclick: () => login.open() }, 'Log in'));
    }
    if (canPost) items.push(h('button', { type: 'button', class: 'btn btn--primary', onclick: () => compose.open() }, 'New event'));
    root.replaceChildren(...items);
    render(); // edit rights may have changed
  }

  setupTabs(store);
  store.subscribe((s, prev) => {
    if (s.tab !== prev.tab) showTab(s.tab);
    render();
  });
  window.addEventListener('online', () => {
    if (store.get().stale || store.get().error) loadEvents();
    subs.reconnectNow();
  });

  try {
    const { data } = await api.config();
    store.set({ config: data });
    if (data?.devMode) devWriter = createDevWriter({ api });
  } catch { /* offline with nothing cached: read-only */ }
  renderAccount();
  loadFacets();
  loadEvents();
  subs.load();
  registerServiceWorker();
}

function dedupe(events) {
  const seen = new Set();
  return events.filter((e) => !seen.has(e.uri) && seen.add(e.uri));
}

function setupTabs(store) {
  const tabs = [...document.querySelectorAll('[role=tab]')];
  tabs.forEach((tab, i) => {
    tab.addEventListener('click', () => store.set({ tab: tab.dataset.tab }));
    tab.addEventListener('keydown', (e) => {
      const d = { ArrowRight: 1, ArrowLeft: -1 }[e.key];
      if (!d) return;
      const next = tabs[(i + d + tabs.length) % tabs.length];
      next.focus();
      store.set({ tab: next.dataset.tab });
    });
  });
}

function showTab(name) {
  for (const tab of document.querySelectorAll('[role=tab]')) {
    const on = tab.dataset.tab === name;
    tab.setAttribute('aria-selected', String(on));
    tab.tabIndex = on ? 0 : -1;
    document.getElementById(tab.getAttribute('aria-controls')).hidden = !on;
  }
}

function registerServiceWorker() {
  if (!('serviceWorker' in navigator)) return;
  navigator.serviceWorker.register('./sw.js').catch(() => { /* offline support is best-effort */ });
}

/**
 * Minimal AT Protocol client (no SDK): login via app password and write records to the
 * user's own PDS. Session lives in memory + sessionStorage (cleared when the tab closes).
 * @module atproto
 */
import { EVENT_NSID, LISTENER_NSID, buildListenerRecord } from './records.js';
import { getStorage, storageGet, storageSet } from './util.js';

const SESSION_KEY = 'cm:atproto-session';

/** Error from an XRPC call. */
export class XrpcError extends Error {
  /** @param {number} status @param {string} error @param {string} [message] */
  constructor(status, error, message) {
    super(message || error || `XRPC ${status}`);
    this.name = 'XrpcError';
    this.status = status;
    this.error = error;
  }
}

/**
 * Resolve the PDS endpoint from a DID document.
 * @param {object} doc
 * @returns {string|null}
 */
export function pdsFromDidDoc(doc) {
  const svc = (doc?.service || []).find(
    (s) => (s.id === '#atproto_pds' || s.id?.endsWith('#atproto_pds')) && s.type === 'AtprotoPersonalDataServer',
  );
  return svc?.serviceEndpoint ? String(svc.serviceEndpoint).replace(/\/$/, '') : null;
}

/**
 * URL of a DID document for did:plc / did:web.
 * @param {string} did
 * @param {string} [plcBase]
 * @returns {string}
 */
export function didDocUrl(did, plcBase = 'https://plc.directory') {
  if (did.startsWith('did:plc:')) return `${plcBase}/${did}`;
  if (did.startsWith('did:web:')) {
    const [host, ...path] = did.slice(8).split(':').map(decodeURIComponent);
    return path.length ? `https://${host}/${path.join('/')}/did.json` : `https://${host}/.well-known/did.json`;
  }
  throw new Error(`Unsupported DID method: ${did}`);
}

/**
 * Create the PDS writer.
 * @param {{fetchImpl?: typeof fetch, storage?: Storage|null,
 *   resolverBases?: string[], plcBase?: string}} [opts]
 */
export function createAtprotoWriter({
  fetchImpl = (...a) => fetch(...a),
  storage = getStorage('sessionStorage'),
  resolverBases = ['https://public.api.bsky.app', 'https://bsky.social'],
  plcBase = 'https://plc.directory',
} = {}) {
  /** @type {{did: string, handle: string, accessJwt: string, refreshJwt: string, pds: string}|null} */
  let session = null;
  try {
    session = JSON.parse(storageGet(storage, SESSION_KEY) || 'null');
  } catch {
    session = null;
  }

  async function json(res) {
    const text = await res.text();
    return text ? JSON.parse(text) : {};
  }

  async function call(url, { method = 'GET', body, token } = {}) {
    const headers = { Accept: 'application/json' };
    if (body !== undefined) headers['Content-Type'] = 'application/json';
    if (token) headers.Authorization = `Bearer ${token}`;
    const res = await fetchImpl(url, { method, headers, body: body === undefined ? undefined : JSON.stringify(body) });
    const data = await json(res).catch(() => ({}));
    if (!res.ok) throw new XrpcError(res.status, data.error, data.message);
    return data;
  }

  async function resolveHandle(handle) {
    let lastErr;
    for (const base of resolverBases) {
      try {
        const q = encodeURIComponent(handle);
        const { did } = await call(`${base}/xrpc/com.atproto.identity.resolveHandle?handle=${q}`);
        if (did) return did;
      } catch (err) {
        lastErr = err;
      }
    }
    throw lastErr || new Error(`Could not resolve handle ${handle}`);
  }

  async function resolvePds(did) {
    const doc = await call(didDocUrl(did, plcBase));
    const pds = pdsFromDidDoc(doc);
    if (!pds) throw new Error('No PDS found in DID document');
    return pds;
  }

  function save(next) {
    session = next;
    storageSet(storage, SESSION_KEY, next ? JSON.stringify(next) : null);
  }

  async function refresh() {
    const data = await call(`${session.pds}/xrpc/com.atproto.server.refreshSession`, {
      method: 'POST',
      token: session.refreshJwt,
    });
    save({ ...session, accessJwt: data.accessJwt, refreshJwt: data.refreshJwt, handle: data.handle || session.handle });
  }

  const isExpired = (err) => err instanceof XrpcError && (err.status === 401 || err.status === 400) && err.error === 'ExpiredToken';

  async function xrpcPost(nsid, body) {
    if (!session) throw new Error('Not logged in');
    const go = () => call(`${session.pds}/xrpc/${nsid}`, { method: 'POST', body, token: session.accessJwt });
    try {
      return await go();
    } catch (err) {
      if (!isExpired(err)) throw err;
      await refresh();
      return go();
    }
  }

  const uriOf = (collection, rkey) => `at://${session.did}/${collection}/${rkey}`;

  return {
    kind: 'atproto',
    /** @returns {string|null} logged-in DID */
    get did() {
      return session?.did ?? null;
    },
    /** @returns {string|null} logged-in handle */
    get handle() {
      return session?.handle ?? null;
    },
    /** @returns {boolean} */
    get loggedIn() {
      return Boolean(session);
    },
    /**
     * Log in: handle → DID → DID doc → PDS → createSession.
     * @param {string} identifier handle (e.g. alice.bsky.social) or DID
     * @param {string} appPassword
     * @returns {Promise<{did: string, handle: string}>}
     */
    async login(identifier, appPassword) {
      const id = identifier.trim().replace(/^@/, '');
      const did = id.startsWith('did:') ? id : await resolveHandle(id);
      const pds = await resolvePds(did);
      const data = await call(`${pds}/xrpc/com.atproto.server.createSession`, {
        method: 'POST',
        body: { identifier: id, password: appPassword },
      });
      save({ did: data.did, handle: data.handle, accessJwt: data.accessJwt, refreshJwt: data.refreshJwt, pds });
      return { did: data.did, handle: data.handle };
    },
    /** Log out (best-effort server-side deleteSession) and clear local session. */
    async logout() {
      const s = session;
      save(null);
      if (s) {
        await call(`${s.pds}/xrpc/com.atproto.server.deleteSession`, { method: 'POST', token: s.refreshJwt }).catch(() => {});
      }
    },
    /**
     * @param {object} record app.communitymap.event record
     * @returns {Promise<{uri: string, cid: string}>}
     */
    createEvent(record) {
      return xrpcPost('com.atproto.repo.createRecord', {
        repo: session?.did,
        collection: EVENT_NSID,
        record: { ...record, $type: EVENT_NSID },
      });
    },
    /**
     * @param {string} rkey @param {object} record
     * @returns {Promise<{uri: string, cid: string}>}
     */
    updateEvent(rkey, record) {
      return xrpcPost('com.atproto.repo.putRecord', {
        repo: session?.did,
        collection: EVENT_NSID,
        rkey,
        record: { ...record, $type: EVENT_NSID },
      });
    },
    /** @param {string} rkey @returns {Promise<{uri: string}>} */
    async deleteEvent(rkey) {
      await xrpcPost('com.atproto.repo.deleteRecord', { repo: session?.did, collection: EVENT_NSID, rkey });
      return { uri: uriOf(EVENT_NSID, rkey) };
    },
    /**
     * @param {string} eventUri @param {string} hashtag
     * @returns {Promise<{uri: string, cid: string}>}
     */
    createListener(eventUri, hashtag) {
      return xrpcPost('com.atproto.repo.createRecord', {
        repo: session?.did,
        collection: LISTENER_NSID,
        record: buildListenerRecord(eventUri, hashtag),
      });
    },
  };
}

/**
 * Small localStorage-backed response cache. Every storage access is guarded.
 * @module cache
 */
import { getStorage, storageGet, storageSet } from './util.js';

/**
 * @typedef {{value: any, savedAt: number}} CacheEntry
 * @typedef {{get(key: string): CacheEntry|null, set(key: string, value: any): boolean,
 *            remove(key: string): void, clear(): void}} Cache
 */

/**
 * Create a cache.
 * @param {{storage?: Storage|null, prefix?: string, now?: () => number}} [opts]
 * @returns {Cache}
 */
export function createCache({ storage = getStorage('localStorage'), prefix = 'cm:cache:', now = Date.now } = {}) {
  const keyOf = (k) => prefix + k;

  function get(key) {
    const raw = storageGet(storage, keyOf(key));
    if (!raw) return null;
    try {
      const entry = JSON.parse(raw);
      return entry && 'value' in entry ? entry : null;
    } catch {
      return null;
    }
  }

  function set(key, value) {
    const raw = JSON.stringify({ value, savedAt: now() });
    if (storageSet(storage, keyOf(key), raw)) return true;
    clear(); // probably quota: drop our old entries and retry once
    return storageSet(storage, keyOf(key), raw);
  }

  function remove(key) {
    storageSet(storage, keyOf(key), null);
  }

  function clear() {
    try {
      for (let i = storage.length - 1; i >= 0; i--) {
        const k = storage.key(i);
        if (k && k.startsWith(prefix)) storage.removeItem(k);
      }
    } catch {
      /* storage unavailable */
    }
  }

  return { get, set, remove, clear };
}

/**
 * "Demo mode" writer: same interface as the atproto writer, but posts Jetstream-shaped
 * commits to POST /api/dev/records under a fake, locally persisted DID.
 * @module devWriter
 */
import { EVENT_NSID, LISTENER_NSID, buildListenerRecord } from './records.js';
import { getStorage, randomId, storageGet, storageSet } from './util.js';

const DID_KEY = 'cm:demo-did';
const S32 = '234567abcdefghijklmnopqrstuvwxyz';

let lastTidUs = 0n;
const clockId = BigInt(Math.floor(Math.random() * 1024));

/**
 * Generate a TID-like record key: 13 chars of base32-sortable encoding of
 * (microsecond timestamp << 10 | clock id). Monotonic within this page.
 * @param {number} [nowMs]
 * @returns {string}
 */
export function generateTid(nowMs = Date.now()) {
  let us = BigInt(Math.floor(nowMs)) * 1000n;
  if (us <= lastTidUs) us = lastTidUs + 1n;
  lastTidUs = us;
  let n = ((us & ((1n << 53n) - 1n)) << 10n) | clockId;
  let out = '';
  for (let i = 0; i < 13; i++) {
    out = S32[Number(n & 31n)] + out;
    n >>= 5n;
  }
  return out;
}

/**
 * Build a Jetstream-shaped commit (see CONTRACTS.md).
 * @param {{did: string, operation: 'create'|'update'|'delete', collection: string,
 *   rkey: string, record?: object, nowMs?: number}} p
 * @returns {object}
 */
export function buildCommit({ did, operation, collection, rkey, record, nowMs = Date.now() }) {
  const commit = { operation, collection, rkey };
  if (operation !== 'delete') {
    commit.cid = `bafydev${randomId(24)}`;
    commit.record = record;
  }
  return { did, time_us: Math.floor(nowMs) * 1000, kind: 'commit', commit };
}

/**
 * @param {{api: {postDevRecord(c: object): Promise<any>}, storage?: Storage|null}} opts

export function createDevWriter({ api, storage = getStorage('localStorage') }) {
  let did = storageGet(storage, DID_KEY);

    storageSet(storage, DID_KEY, did);

  async function commit(operation, collection, rkey, record) {
    const c = buildCommit({ did, operation, collection, rkey, record });
    await api.postDevRecord(c);

  }

  return {

const PROPS = new Set(['value', 'checked', 'selected', 'disabled', 'hidden', 'open']);

/**
 * Create an element. Strings/numbers become text nodes (auto-escaped).
 * @param {string} tag
 * @param {Object<string, any>} [props] attributes; `class`, `dataset`, `onX` handlers supported
 * @param {...any} children nodes, strings, arrays, or null/false (skipped)
 * @returns {HTMLElement}
 */
export function h(tag, props, ...children) {
  const el = document.createElement(tag);
  for (const [key, val] of Object.entries(props || {})) setProp(el, key, val);
  append(el, children);
  return el;
}

function setProp(el, key, val) {
  if (val == null || val === false) return;
  if (key === 'class') el.className = val;
  else if (key === 'dataset') Object.assign(el.dataset, val);
  else if (key.startsWith('on') && typeof val === 'function') {
    el.addEventListener(key.slice(2).toLowerCase(), val);
  } else if (PROPS.has(key)) el[key] = val;
  else el.setAttribute(key, val === true ? '' : String(val));
}

/**
 * Append children (flattening arrays, skipping null/false).
 * @param {Node} el
 * @param {any[]} children
 */
export function append(el, children) {
  for (const c of children.flat(Infinity)) {
    if (c == null || c === false) continue;
    el.append(c instanceof Node ? c : document.createTextNode(String(c)));
  }
}

/**
 * Parse an event's string lat/lng into numbers, or null if absent/invalid.
 * @param {{location?: {lat?: string, lng?: string}}} event
 * @returns {{lat: number, lng: number}|null}
 */
export function coordsOf(event) {
  const loc = event?.location;
  if (!loc || loc.lat == null || loc.lng == null || loc.lat === '' || loc.lng === '') return null;
  const lat = Number(loc.lat);
  const lng = Number(loc.lng);
  if (!Number.isFinite(lat) || !Number.isFinite(lng)) return null;
  if (Math.abs(lat) > 90 || Math.abs(lng) > 180) return null;
  return { lat, lng };
}

/**
 * Great-circle distance in km (haversine).
 * @param {{lat: number, lng: number}} a
 * @param {{lat: number, lng: number}} b
 * @returns {number}
 */
export function haversineKm(a, b) {
  const R = 6371;
  const rad = (d) => (d * Math.PI) / 180;
  const dLat = rad(b.lat - a.lat);
  const dLng = rad(b.lng - a.lng);
  const s = Math.sin(dLat / 2) ** 2 + Math.cos(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.sin(dLng / 2) ** 2;
  return 2 * R * Math.asin(Math.min(1, Math.sqrt(s)));
}

/**
 * Sort events nearest-first from `origin`; events without coordinates go last (stable).
 * Returns copies annotated with `distanceKm` (null when unknown).
 * @param {object[]} events
 * @param {{lat: number, lng: number}} origin
 * @returns {object[]}
 */
export function sortByDistance(events, origin) {
  return events
    .map((e, i) => {
      const c = coordsOf(e);
      return { e: { ...e, distanceKm: c ? haversineKm(origin, c) : null }, i };
    })
    .sort((x, y) => {
      const dx = x.e.distanceKm ?? Infinity;
      const dy = y.e.distanceKm ?? Infinity;
      return dx === dy ? x.i - y.i : dx - dy;
    })
    .map((x) => x.e);
}

/**
 * Local-date key "YYYY-MM-DD".
 * @param {Date} d
 * @returns {string}
 */
export function dayKey(d) {
  const p = (n) => String(n).padStart(2, '0');
  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`;
}

/**
 * Human label for a day relative to `now`: Today, Tomorrow, weekday (within a week), or date.
 * @param {Date} day
 * @param {Date} now
 * @param {string} [locale]
 * @returns {string}
 */
export function dayLabel(day, now, locale) {
  const startOf = (d) => new Date(d.getFullYear(), d.getMonth(), d.getDate());
  const diff = Math.round((startOf(day) - startOf(now)) / 86400000);
  if (diff <= 0) return 'Today';
  if (diff === 1) return 'Tomorrow';
  if (diff < 7) return new Intl.DateTimeFormat(locale, { weekday: 'long' }).format(day);
  return new Intl.DateTimeFormat(locale, { weekday: 'short', month: 'short', day: 'numeric' }).format(day);
}

/**
 * Group events by local day. Events already in progress are grouped under today.
 * Groups are chronological; order within a group is preserved (e.g. distance order).
 * @param {object[]} events
 * @param {{now?: Date, locale?: string}} [opts]
 * @returns {{key: string, label: string, events: object[]}[]}
 */
export function groupByDay(events, { now = new Date(), locale } = {}) {
  const groups = new Map();
  for (const ev of events) {
    const start = new Date(ev.startsAt);
    const day = start < now ? now : start;
    const key = dayKey(day);
    if (!groups.has(key)) groups.set(key, { key, label: dayLabel(day, now, locale), events: [] });
    groups.get(key).events.push(ev);
  }
  return [...groups.values()].sort((a, b) => (a.key < b.key ? -1 : a.key > b.key ? 1 : 0));
}

/**
 * True when the event was edited after it was first indexed.
 * @param {{updatedAt?: string, indexedAt?: string}} event
 * @returns {boolean}
 */
export function isUpdated(event) {
  if (!event?.updatedAt || !event?.indexedAt) return false;
  return new Date(event.updatedAt) > new Date(event.indexedAt);
}

/**
 * @param {{geolocation?: Geolocation|null, timeoutMs?: number}} [opts]
 */
export function createGeo({ geolocation = globalThis.navigator?.geolocation ?? null, timeoutMs = 10000 } = {}) {
  /** @type {{lat: number, lng: number}|null} */
  let position = null;

  return {
    /** @returns {boolean} whether the browser supports geolocation */
    get supported() {
      return Boolean(geolocation);
    },
    /** @returns {{lat: number, lng: number}|null} last known position (memory only) */
    get position() {
      return position;
    },
    /**
     * Ask the browser for the current position. Call only from a user gesture.
     * @returns {Promise<{lat: number, lng: number}>}
     */
    request() {
      if (!geolocation) return Promise.reject(new Error('Location is not available on this device'));
      return new Promise((resolve, reject) => {
        geolocation.getCurrentPosition(
          (p) => {
            position = { lat: p.coords.latitude, lng: p.coords.longitude };
            resolve(position);
          },
          (err) => reject(new Error(err?.message || 'Could not get your location')),
          { enableHighAccuracy: false, maximumAge: 60000, timeout: timeoutMs },
        );
      });
    },
    /** Forget the position. */
    clear() {
      position = null;
    },
  };
}

/**

 * omitted; lat/lng become strings; `createdAt` defaults to now.

/**

 * Field keys: name, description, startsAt, endsAt, locationName, address, coords, tags, group.
 * @param {object} r event record
 * @returns {Record<string, string>}
 */
export function validateEventRecord(r) {
  const e = {};
  if (!r.name) e.name = 'Name is required.';
  else if (bytes(r.name) > 200) e.name = 'Name is too long (max 200).';
  if (r.description && bytes(r.description) > 3000) e.description = 'Description is too long (max 3000).';
  if (!isDate(r.startsAt)) e.startsAt = 'Start time is required.';
  if (!isDate(r.endsAt)) e.endsAt = 'End time is required.';
  else if (isDate(r.startsAt) && Date.parse(r.endsAt) < Date.parse(r.startsAt)) {
    e.endsAt = 'End must be at or after start.';
  }
  const loc = r.location || {};
  if (!loc.name) e.locationName = 'Location name is required.';
  else if (bytes(loc.name) > 200) e.locationName = 'Location name is too long (max 200).';
  if (loc.address && bytes(loc.address) > 300) e.address = 'Address is too long (max 300).';
  const coordsErr = validateCoords(loc.lat, loc.lng);
  if (coordsErr) e.coords = coordsErr;
  const tags = r.tags || [];
  if (tags.length > 10) e.tags = 'At most 10 tags.';
  else if (tags.some((t) => bytes(t) > 32)) e.tags = 'Each tag must be 32 characters or fewer.';
  if (r.group && bytes(r.group) > 100) e.group = 'Group is too long (max 100).';
  return e;
}

/**

 * @module store
 */

/**
 * @template T
 * @typedef {{get(): T, set(patch: Partial<T> | ((s: T) => Partial<T>)): void,
 *            subscribe(fn: (state: T, prev: T) => void): () => void}} Store
 */

/**
 * Create a store holding a plain object. `set` shallow-merges and notifies subscribers.
 * @template {object} T
 * @param {T} initial
 * @returns {Store<T>}
 */
export function createStore(initial) {
  let state = { ...initial };
  const listeners = new Set();

  return {
    get: () => state,
    set(patch) {
      const prev = state;
      const delta = typeof patch === 'function' ? patch(prev) : patch;
      state = { ...prev, ...delta };
      for (const fn of [...listeners]) fn(state, prev);
    },
    subscribe(fn) {
      listeners.add(fn);
      return () => listeners.delete(fn);
    },
  };
}

/**

/**
 * Pure catch-up merge: keep notifications newer than `lastId`, sorted by id, deduped.
 * @param {number|null} lastId
 * @param {{id: number}[]} list
 * @returns {{fresh: object[], lastId: number|null}}
 */
export function mergeNotifications(lastId, list) {
  const seen = new Set();
  const fresh = (list || [])
    .filter((n) => typeof n.id === 'number' && (lastId == null || n.id > lastId))
    .filter((n) => !seen.has(n.id) && seen.add(n.id))
    .sort((a, b) => a.id - b.id);
  const next = fresh.length ? fresh[fresh.length - 1].id : lastId;
  return { fresh, lastId: next };
}

  }

  function setLast(id) {
    lastId = id;
    storageSet(storage, LAST_KEY, id == null ? null : String(id));
  }

  function deliver(n) {
    toast(n.message);
    const hidden = globalThis.document?.hidden;
    if (hidden && NotificationImpl?.permission === 'granted') {
      try {
        new NotificationImpl('Community Map', { body: n.message, tag: `cm-${n.id}` });
      } catch {
        /* some platforms only allow notifications from a service worker */
      }
    }
    onNotification(n);
  }

  function handle(list) {
    const { fresh, lastId: next } = mergeNotifications(lastId, list);
    setLast(next);
    fresh.forEach(deliver);
  }

  /** Fetch notifications missed while disconnected. First run only records a baseline. */
  async function catchUp() {
    try {
      const { data } = await api.notifications(subscriberId, lastId);
      if (lastId == null) {
        setLast(mergeNotifications(null, data).lastId ?? 0);
        return;
      }
      handle(data);
    } catch {
      /* offline; next reconnect retries */
    }
  }

  function connect() {
    if (!EventSourceImpl || source || subscribed.size === 0) return;
    clearTimeout(timer);
    source = new EventSourceImpl(api.notificationStreamUrl(subscriberId));
    source.onopen = () => {
      attempts = 0;
      catchUp();
    };
    source.onmessage = (ev) => {
      try {
        handle([JSON.parse(ev.data)]);
      } catch {
        /* ignore malformed message */
      }
    };
    source.onerror = () => {
      disconnect();
      timer = setTimeout(connect, backoffMs(attempts++, baseDelayMs));
    };
  }

  function disconnect() {
    clearTimeout(timer);
    if (source) source.close();
    source = null;
  }

  async function askPermissionOnce() {
    if (!NotificationImpl || NotificationImpl.permission !== 'default') return;
    if (storageGet(storage, ASKED_KEY)) return;
    storageSet(storage, ASKED_KEY, '1');
    try {
      await NotificationImpl.requestPermission();
    } catch {
      /* ignore */
    }
  }

  function changed() {
    onChange(new Set(subscribed));
    if (subscribed.size) connect();
    else disconnect();
  }

  return {
    subscriberId,
    /** @returns {Set<string>} copy of subscribed event URIs */
    get subscribed() {
      return new Set(subscribed);
    },
    /** @param {string} uri @returns {boolean} */
    has: (uri) => subscribed.has(uri),
    /** Load the server's subscription list and connect the stream if needed. */
    async load() {
      try {
        const { data } = await api.listSubscriptions(subscriberId);
        subscribed.clear();
        (data || []).forEach((u) => subscribed.add(u));
      } catch {
        /* offline with no cache */
      }
      changed();
    },
    /**
     * Toggle a subscription. Asks for Notification permission on the very first subscribe.
     * @param {string} uri
     * @returns {Promise<boolean>} new subscribed state
     */
    async toggle(uri) {
      if (subscribed.has(uri)) {
        await api.unsubscribe(subscriberId, uri);
        subscribed.delete(uri);
      } else {
        await api.subscribe(subscriberId, uri);
        subscribed.add(uri);
        askPermissionOnce();
      }
      changed();
      return subscribed.has(uri);
    },
    /** Reconnect immediately (e.g. on the window "online" event). */
    reconnectNow() {
      attempts = 0;
      disconnect();
      connect();
    },
    catchUp,
    disconnect,
  };
}

 * Small generic utilities.
 * @module util

/**
 * Debounce a function.
 * @template {(...a: any[]) => void} F
 * @param {F} fn
 * @param {number} ms
 * @returns {F & {cancel: () => void}}
 */
export function debounce(fn, ms) {
  let t;
  const wrapped = (...args) => {
    clearTimeout(t);
    t = setTimeout(() => fn(...args), ms);
  };
  wrapped.cancel = () => clearTimeout(t);
  return wrapped;
}

/**
 * Random lowercase base32-ish id (uses crypto when available).
 * @param {number} [len=16]
 * @returns {string}
 */
export function randomId(len = 16) {
  const alphabet = 'abcdefghijklmnopqrstuvwxyz234567';
  const bytes = new Uint8Array(len);
  if (globalThis.crypto?.getRandomValues) globalThis.crypto.getRandomValues(bytes);
  else for (let i = 0; i < len; i++) bytes[i] = Math.floor(Math.random() * 256);
  return Array.from(bytes, (b) => alphabet[b % 32]).join('');
}

/**
 * Safely obtain a Web Storage object (access itself can throw in private modes).
 * @param {'localStorage'|'sessionStorage'} name
 * @returns {Storage|null}
 */
export function getStorage(name) {
  try {
    return globalThis[name] ?? null;
  } catch {
    return null;
  }
}

/**
 * Storage read that never throws.
 * @param {Storage|null} storage
 * @param {string} key
 * @returns {string|null}
 */
export function storageGet(storage, key) {
  try {
    return storage ? storage.getItem(key) : null;
  } catch {
    return null;
  }
}

/**
 * Storage write that never throws. Returns false if it failed.
 * @param {Storage|null} storage
 * @param {string} key
 * @param {string|null} value null removes the key
 * @returns {boolean}
 */
export function storageSet(storage, key, value) {
  try {
    if (!storage) return false;
    if (value == null) storage.removeItem(key);
    else storage.setItem(key, value);
    return true;
  } catch {
    return false;
  }
}

/**
 * ISO timestamp → value for <input type="datetime-local"> in local time.
 * @param {string} iso
 * @returns {string}
 */
export function isoToLocalInput(iso) {
  if (!iso) return '';
  const d = new Date(iso);
  const p = (n) => String(n).padStart(2, '0');
  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}T${p(d.getHours())}:${p(d.getMinutes())}`;
}

/**
 * datetime-local value (local time) → ISO string, or "" if empty/invalid.
 * @param {string} value
 * @returns {string}
 */
export function localInputToIso(value) {
  if (!value) return '';
  const d = new Date(value);
  return Number.isNaN(d.getTime()) ? '' : d.toISOString();
}

const FIELDS = ['name', 'description', 'startsAt', 'endsAt', 'locationName', 'address', 'lat', 'lng', 'tags', 'group', 'hashtag'];

/**
 * Read raw values from the compose form.
 * @param {HTMLFormElement} form
 * @returns {Record<string, string>}
 */
export function readForm(form) {
  return Object.fromEntries(FIELDS.map((f) => [f, form.elements[f]?.value ?? '']));
}

/**
 * Turn form values into a record and validate it.
 * Error keys are form field names (coords errors are reported on "lat").
 * @param {Record<string, string>} values
 * @param {{createdAt?: string, sourceUrl?: string}} [base] preserved fields when editing
 * @returns {{record: object, errors: Record<string, string>}}
 */
export function validateForm(values, base = {}) {
  const record = buildEventRecord({
    ...values,
    startsAt: localInputToIso(values.startsAt),
    endsAt: localInputToIso(values.endsAt),
    createdAt: base.createdAt,
    sourceUrl: base.sourceUrl,
  });
  const errors = validateEventRecord(record);
  const partialCoords = Boolean(values.lat?.trim()) !== Boolean(values.lng?.trim());
  if (partialCoords) errors.coords = 'Enter both latitude and longitude, or neither.';
  if (errors.coords) {
    errors.lat = errors.coords;
    delete errors.coords;
  }
  const tagErr = validateHashtag(values.hashtag);
  if (tagErr) errors.hashtag = tagErr;
  return { record, errors };
}

function field(name, label, input, hint) {

  );
}

/**
 * Show errors next to fields (aria-invalid + aria-describedby) and focus the first one.
 * @param {HTMLFormElement} form
 * @param {Record<string, string>} errors
 */
export function showErrors(form, errors) {
  for (const f of FIELDS) {
    const input = form.elements[f];
    const err = form.querySelector(`#c-${f}-err`);
    if (!input || !err) continue;
    const msg = errors[f] || '';
    err.textContent = msg;
    input.setAttribute('aria-invalid', String(Boolean(msg)));
    const hint = form.querySelector(`#c-${f}-hint`) ? `c-${f}-hint ` : '';
    input.setAttribute('aria-describedby', `${hint}c-${f}-err`.trim());
  }
  const first = FIELDS.find((f) => errors[f]);
  if (first) form.elements[first].focus();
}

/**
 * Create the compose dialog controller.
 * @param {{getWriter: () => object|null, geo?: {request(): Promise<{lat: number, lng: number}>},
 *   onSaved?: (r: {uri: string}) => void, onDeleted?: (uri: string) => void,
 *   toast?: (msg: string, o?: object) => void, confirmImpl?: (msg: string) => boolean}} deps
 * @returns {{open(ev?: object|null): void, remove(ev: object): Promise<void>, element: HTMLDialogElement}}
 */
export function createComposeView({ getWriter, geo, onSaved, onDeleted, toast = () => {}, confirmImpl = (m) => globalThis.confirm(m) }) {
  const dialog = h('dialog', { class: 'dialog', 'aria-labelledby': 'compose-title' });

  function open(ev = null) {
    const form = buildForm(ev);
    const status = form.querySelector('.form-status');
    form.addEventListener('submit', (e) => {
      e.preventDefault();
      submit(form, ev, status);
    });
    form.querySelector('[data-action=cancel]').onclick = () => closeDialog(dialog);
    form.querySelector('[data-action=locate]').onclick = () => locate(form, status);
    const del = form.querySelector('[data-action=delete]');
    if (del) del.onclick = () => remove(ev);
    dialog.replaceChildren(h('h2', { id: 'compose-title' }, ev ? 'Edit event' : 'New event'), form);
    openDialog(dialog);
  }

  async function locate(form, status) {
    try {
      const p = await geo.request();
      form.elements.lat.value = p.lat.toFixed(6);
      form.elements.lng.value = p.lng.toFixed(6);
    } catch (err) {
      status.textContent = err.message;
    }
  }

  async function submit(form, ev, status) {
    const values = readForm(form);
    const { record, errors } = validateForm(values, ev ? { createdAt: ev.createdAt ?? ev.indexedAt, sourceUrl: ev.sourceUrl } : {});
    showErrors(form, errors);
    if (Object.keys(errors).length) return;
    const writer = getWriter();
    if (!writer) return void (status.textContent = 'Log in to post events.');
    const btn = form.querySelector('[type=submit]');
    btn.disabled = true;
    status.textContent = 'Saving…';
    try {
      const res = ev ? await writer.updateEvent(parseAtUri(ev.uri).rkey, record) : await writer.createEvent(record);
      if (values.hashtag.trim()) await writer.createListener(res.uri, values.hashtag);
      closeDialog(dialog);
      onSaved?.(res);
    } catch (err) {
      status.textContent = `Could not save: ${err.message}`;
    } finally {
      btn.disabled = false;
    }
  }

  async function remove(ev) {
    const writer = getWriter();
    if (!writer || !confirmImpl(`Delete “${ev.name}”?`)) return;
    try {
      await writer.deleteEvent(parseAtUri(ev.uri).rkey);
      closeDialog(dialog);
      onDeleted?.(ev.uri);
    } catch (err) {
      toast(`Could not delete: ${err.message}`, { kind: 'error' });
    }
  }

  return { open, remove, element: dialog };
}

/**
 * Filter bar: debounced text search, group select, tag chips, "Near me" toggle.
 * Stateless UI: it renders a FilterState and reports changes via onChange(patch).
 * @module views/filterBar
 */
import { h } from '../dom.js';
import { toggleTag } from '../filters.js';
import { debounce } from '../util.js';

/**
 * @param {HTMLElement} root
 * @param {{onChange: (patch: Partial<import('../filters.js').FilterState>) => void,
 *   debounceMs?: number, geoSupported?: boolean}} opts
 * @returns {{render(state: import('../filters.js').FilterState,
 *   options: {tags: {value: string, count: number}[], groups: {value: string, count: number}[],
 *   nearMeBusy?: boolean}): void}}
 */
export function createFilterBar(root, { onChange, debounceMs = 300, geoSupported = true }) {
  let state = { tags: [], group: '', q: '', nearMe: false };
  const emitQ = debounce((q) => onChange({ q }), debounceMs);

  const search = h('input', {
    type: 'search', id: 'f-q', name: 'q', autocomplete: 'off', placeholder: 'Search events',
    oninput: (e) => emitQ(e.target.value),
  });
  const group = h('select', { id: 'f-group', name: 'group', onchange: (e) => onChange({ group: e.target.value }) });
  const nearMe = h('button', {
    type: 'button', class: 'btn', 'aria-pressed': 'false', hidden: !geoSupported,
    onclick: () => onChange({ nearMe: !state.nearMe }),
  }, 'Near me');
  const chips = h('ul', { class: 'chips', 'aria-label': 'Filter by tag' });

  root.replaceChildren(
    h('div', { class: 'filters__row' },
      h('label', { for: 'f-q', class: 'visually-hidden' }, 'Search events'), search,
      h('label', { for: 'f-group', class: 'visually-hidden' }, 'Group'), group,
      nearMe,
    ),
    chips,
  );

  function render(next, { tags = [], groups = [], nearMeBusy = false } = {}) {
    state = next;
    if (document.activeElement !== search && search.value !== next.q) search.value = next.q;
    renderGroups(groups, next.group);
    nearMe.setAttribute('aria-pressed', String(next.nearMe));
    nearMe.textContent = nearMeBusy ? 'Locating…' : 'Near me';
    renderChips(tags, next.tags);
  }

  function renderGroups(groups, selected) {
    const opts = [h('option', { value: '' }, 'All groups'),
      ...groups.map((g) => h('option', { value: g.value }, `${g.value} (${g.count})`))];
    if (selected && !groups.some((g) => g.value === selected)) opts.push(h('option', { value: selected }, selected));
    group.replaceChildren(...opts);
    group.value = selected || '';
  }

  function renderChips(tags, active) {
    const focused = document.activeElement?.dataset?.tag;
    const values = [...new Set([...active, ...tags.map((t) => t.value)])];
    chips.replaceChildren(...values.map((v) => h('li', null,
      h('button', {
        type: 'button', class: 'chip', 'aria-pressed': String(active.includes(v)), dataset: { tag: v },
        onclick: () => onChange({ tags: toggleTag(state.tags, v) }),
      }, `#${v}`),
    )));
    if (focused) [...chips.querySelectorAll('button')].find((b) => b.dataset.tag === focused)?.focus();
  }

  return { render };
}

    ev.tags?.length && h('ul', { class: 'chips', 'aria-label': 'Tags' },

          onclick: () => onTagClick?.(t),
        }, `#${t}`),
      )),
    ),
    h('div', { class: 'card__actions' },
      h('button', {
        type: 'button', class: `btn ${isSub ? 'btn--on' : ''}`, 'aria-pressed': String(isSub),
        dataset: { focusKey: `sub:${ev.uri}` }, onclick: () => onToggleSubscribe?.(ev.uri),
      }, isSub ? 'Subscribed' : 'Subscribe'),
    ),
  );
}

function hashId(s) {
  let x = 0;
  for (let i = 0; i < s.length; i++) x = (x * 31 + s.charCodeAt(i)) | 0;
  return (x >>> 0).toString(36);
}

/**
 * Create the list view.
 * @param {HTMLElement} root
 * @param {{onLoadMore?: () => void, IntersectionObserverImpl?: typeof IntersectionObserver|null}} [opts]
 * @returns {{render(state: ListState): void, destroy(): void}}
 */
export function createListView(root, { onLoadMore, IntersectionObserverImpl = globalThis.IntersectionObserver ?? null } = {}) {
  let loading = false;
  let hasMore = false;
  const more = () => {
    if (!loading && hasMore) onLoadMore?.();
  };
  const observer = IntersectionObserverImpl
    ? new IntersectionObserverImpl((entries) => entries.some((e) => e.isIntersecting) && more(), { rootMargin: '400px' })
    : null;

  function render(state) {
    loading = Boolean(state.loading);
    hasMore = Boolean(state.hasMore);
    const focusKey = document.activeElement?.dataset?.focusKey;
    observer?.disconnect();
    root.setAttribute('aria-busy', String(loading));
    root.replaceChildren(...content(state));
    if (focusKey) [...root.querySelectorAll('[data-focus-key]')].find((el) => el.dataset.focusKey === focusKey)?.focus();
    const sentinel = root.querySelector('.load-more');
    if (sentinel && observer) observer.observe(sentinel);
  }

  function content(state) {
    const nodes = [];
    if (state.stale) nodes.push(staleBanner(state.savedAt));
    if (state.error) nodes.push(h('p', { class: 'notice notice--error', role: 'alert' }, state.error));
    const groups = groupByDay(state.events || [], { now: state.now, locale: state.locale });
    if (!groups.length && !state.loading && !state.error) {
      nodes.push(h('p', { class: 'empty' }, 'No events match your filters. Try clearing a tag or search.'));
    }
    for (const g of groups) {
      const id = `day-${g.key}`;
      nodes.push(h('section', { class: 'day', 'aria-labelledby': id },
        h('h2', { class: 'day__title', id }, g.label),
        h('div', { class: 'grid' }, g.events.map((ev) => renderCard(ev, state))),
      ));
    }
    if (state.loading) nodes.push(h('p', { class: 'loading', role: 'status' }, 'Loading events…'));
    if (state.hasMore) {
      nodes.push(h('button', {
        type: 'button', class: 'btn load-more', disabled: loading, dataset: { focusKey: 'load-more' }, onclick: more,
      }, loading ? 'Loading…' : 'Load more'));
    }
    return nodes;
  }

  function staleBanner(savedAt) {
    const when = savedAt ? ` from ${new Date(savedAt).toLocaleTimeString([], { hour: 'numeric', minute: '2-digit' })}` : '';
    return h('p', { class: 'notice notice--stale', role: 'status' },
      `You're offline or the connection is poor — showing saved events${when}.`);
  }

  return { render, destroy: () => observer?.disconnect() };
}

import { h } from '../dom.js';

/** Default map configuration. Override any field via createMapView({config}). */
export const MAP_CONFIG = Object.freeze({
  leafletJs: 'https://unpkg.com/leaflet@1.9.4/dist/leaflet.js',
  leafletCss: 'https://unpkg.com/leaflet@1.9.4/dist/leaflet.css',
  tileUrl: 'https://tile.openstreetmap.org/{z}/{x}/{y}.png',
  attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors',
  maxZoom: 19,
  center: [32.8801, -117.234], // UCSD
  zoom: 15,
});

/**
 * Pure: events → marker descriptors (only events with valid coordinates).
 * @param {object[]} events
 * @param {{locale?: string, timeZone?: string}} [opts]
 * @returns {{uri: string, lat: number, lng: number, title: string, subtitle: string}[]}
 */
export function buildMarkers(events, opts = {}) {
  return (events || []).flatMap((ev) => {
    const c = coordsOf(ev);
    if (!c) return [];
    const when = formatTimeWindow(ev.startsAt, ev.endsAt, opts);
    return [{ uri: ev.uri, lat: c.lat, lng: c.lng, title: ev.name, subtitle: `${when} · ${ev.location?.name ?? ''}` }];
  });
}

let leafletPromise = null;

/**
 * Inject Leaflet's CSS + script once and resolve with window.L.
 * @param {typeof MAP_CONFIG} config
 * @returns {Promise<any>}
 */
export function loadLeaflet(config = MAP_CONFIG) {
  if (globalThis.L) return Promise.resolve(globalThis.L);
  leafletPromise ??= new Promise((resolve, reject) => {
    document.head.append(h('link', { rel: 'stylesheet', href: config.leafletCss }));
    const script = h('script', { src: config.leafletJs, async: true });
    script.onload = () => resolve(globalThis.L);
    script.onerror = () => {
      leafletPromise = null;
      script.remove();
      reject(new Error('Map library failed to load'));
    };
    document.head.append(script);
  });
  return leafletPromise;
}

/**
 * Create the map view.
 * @param {HTMLElement} root
 * @param {{onOpen?: (uri: string) => void, config?: Partial<typeof MAP_CONFIG>,
 *   load?: (c: typeof MAP_CONFIG) => Promise<any>}} [opts]
 * @returns {{show(events: object[]): Promise<void>}}
 */
export function createMapView(root, { onOpen, config = {}, load = loadLeaflet } = {}) {
  const cfg = { ...MAP_CONFIG, ...config };
  let map = null;
  let layer = null;

  async function ensureMap() {
    if (map) return map;
    root.replaceChildren(h('p', { role: 'status' }, 'Loading map…'));
    const L = await load(cfg);
    const el = h('div', { class: 'map', role: 'region', 'aria-label': 'Event map' });
    root.replaceChildren(el, h('p', { class: 'muted' }, 'The list view has every event and is fully keyboard accessible.'));
    map = L.map(el).setView(cfg.center, cfg.zoom);
    L.tileLayer(cfg.tileUrl, { maxZoom: cfg.maxZoom, attribution: cfg.attribution }).addTo(map);
    layer = L.layerGroup().addTo(map);
    return map;
  }

  function popup(m) {
    return h('div', null,
      h('strong', null, m.title), h('br'), h('span', null, m.subtitle), h('br'),
      h('button', { type: 'button', class: 'btn', onclick: () => onOpen?.(m.uri) }, 'Details'),
    );
  }

  async function show(events) {
    try {
      await ensureMap();
    } catch {
      root.replaceChildren(h('p', { class: 'notice notice--error', role: 'alert' },
        'The map could not load (poor connection?). The list view still works.'));
      return;
    }
    const L = globalThis.L;
    layer.clearLayers();
    const markers = buildMarkers(events);
    for (const m of markers) L.marker([m.lat, m.lng], { title: m.title }).bindPopup(popup(m)).addTo(layer);
    map.invalidateSize();
    if (markers.length) map.fitBounds(markers.map((m) => [m.lat, m.lng]), { maxZoom: 17, padding: [24, 24] });
  }

  return { show };
}

/* Community Map service worker (classic script).
 * - App shell: cache-first.
 * - GET /api/*: network-first, falling back to the cached copy (marked with x-cm-stale: 1).
 * - Never cached: non-GET requests and the SSE notification stream.
 */
const VERSION = 'v1';
const SHELL_CACHE = `cm-shell-${VERSION}`;
const API_CACHE = `cm-api-${VERSION}`;
const SHELL = [
  './', './index.html', './styles.css', './manifest.webmanifest', './icon.svg',
  './src/app.js', './src/api.js', './src/atproto.js', './src/cache.js', './src/devWriter.js',
  './src/dom.js', './src/filters.js', './src/format.js', './src/geo.js', './src/records.js',
  './src/store.js', './src/subscriptions.js', './src/toast.js', './src/util.js',
  './src/views/composeView.js', './src/views/detailView.js', './src/views/filterBar.js',
  './src/views/listView.js', './src/views/loginView.js', './src/views/mapView.js',
];

self.addEventListener('install', (event) => {
  event.waitUntil(caches.open(SHELL_CACHE).then((c) => c.addAll(SHELL)).then(() => self.skipWaiting()));
});

self.addEventListener('activate', (event) => {
  const keep = new Set([SHELL_CACHE, API_CACHE]);
  event.waitUntil(
    caches.keys()
      .then((keys) => Promise.all(keys.filter((k) => k.startsWith('cm-') && !keep.has(k)).map((k) => caches.delete(k))))
      .then(() => self.clients.claim()),
  );
});

self.addEventListener('fetch', (event) => {
  const req = event.request;
  if (req.method !== 'GET') return;
  const url = new URL(req.url);
  if (url.origin !== self.location.origin) return; // Leaflet/tiles: browser default
  const isApi = url.pathname.includes('/api/');
  if (isApi && (url.pathname.endsWith('/api/notifications/stream') || (req.headers.get('accept') || '').includes('text/event-stream'))) return;
  event.respondWith(isApi ? networkFirst(req) : cacheFirst(req));
});

async function networkFirst(req) {
  const cache = await caches.open(API_CACHE);
  try {
    const res = await fetch(req);
    if (res.ok) cache.put(req, res.clone());
    return res;
  } catch (err) {
    const hit = await cache.match(req);
    if (!hit) throw err; // let the page's own cache fallback handle it
    const headers = new Headers(hit.headers);
    headers.set('x-cm-stale', '1');
    return new Response(await hit.blob(), { status: hit.status, statusText: hit.statusText, headers });
  }
}

async function cacheFirst(req) {
  const hit = await caches.match(req, { ignoreSearch: true });
  if (hit) return hit;
  const res = await fetch(req);
  if (res.ok) (await caches.open(SHELL_CACHE)).put(req, res.clone());
  return res;
}

/** In-memory Storage implementation. */
export function memoryStorage() {
  const m = new Map();
  return {
    get length() { return m.size; },
    key: (i) => [...m.keys()][i] ?? null,
    getItem: (k) => (m.has(k) ? m.get(k) : null),
    setItem: (k, v) => m.set(k, String(v)),
    removeItem: (k) => m.delete(k),
    clear: () => m.clear(),
  };
}

/** Minimal fetch Response stand-in. */
export function jsonResponse(body, status = 200, headers = {}) {
  return {
    ok: status >= 200 && status < 300,
    status,
    headers: { get: (k) => headers[k.toLowerCase()] ?? null },
    text: async () => (body === undefined ? '' : JSON.stringify(body)),
  };
}
